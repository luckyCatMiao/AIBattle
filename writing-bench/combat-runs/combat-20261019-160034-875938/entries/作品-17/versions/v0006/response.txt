function updateEnemy(e,dt){
 e.t+=dt;e.flash=Math.max(0,e.flash-dt*4);e.hpShowT-=dt;e.recoil*=Math.exp(-dt*8);
 const dx=P.x-e.x,adx=Math.abs(dx),dirP=dx>=0?1:-1;
 const aliveWep=e.parts.some(p=>p.id==='wep'&&p.alive),aliveOff=e.parts.some(p=>p.id==='off'&&p.alive);
 const legB=e.parts.some(p=>(p.id==='legL'||p.id==='legR')&&!p.alive);
 const legAll=!e.parts.some(p=>(p.id==='legL'||p.id==='legR')&&p.alive);
 const spd=e.def.speed*(1+Math.min(.35,(wave-1)*.035))*(legB&&!legAll?.45:1)*(legAll?0:1);
 let tA=1.25,tB=1.45,sA=7,sB=7,tL=.02;
 if(e.state==='windup'){tA=-2.05;tB=-.5;sA=4.5;sB=5;tL=-.16;}
 else if(e.state==='attack'){tA=.32;tB=.7;sA=24;sB=14;tL=.32;}
 else if(e.state==='hurt'){tA=1.7;tB=1.6;sA=9;sB=9;tL=-.28;}
 else if(e.state==='stagger'){tA=2.5;tB=2.2;sA=4;sB=4;tL=-.4;}
 else if(e.state==='air'){tA=1.9;tB=.6;sA=6;sB=6;tL=.2;}
 e.armA+=clamp(tA-e.armA,-sA*dt,sA*dt);e.armB+=clamp(tB-e.armB,-sB*dt,sB*dt);
 e.lean+=(tL-e.lean)*Math.min(1,dt*6);
 const st=e.state;
 if(st==='spawn'){e.rise=Math.min(1,e.t/.55);
  if(Math.random()<dt*18)inkBurst(e.x+rnd(-14,14),GROUND-rnd(0,e.def.h*e.rise),1,.5);
  if(e.t>.75){e.state='chase';e.t=0;}}
 else if(st==='chase'){
  e.dir=dirP;e.thinkT-=dt;
  let mv=0;
  if(e.type==='gun'){
   if(!aliveWep){if(adx>60)mv=dirP;else if(e.thinkT<=0)startWind(e,'melee');}
   else if(adx<210){
    if((e.x<170&&dirP>0)||(e.x>WORLD_W-170&&dirP<0)){if(e.thinkT<=0)startWind(e,'melee');}
    else mv=-dirP;}
   else if(adx>430)mv=dirP;
   else if(e.thinkT<=0&&atkTokens()<2)startWind(e,'shoot');}
  else{
   const rng=e.def.range*(legAll?.55:1);
   if(adx>rng)mv=dirP;
   else if(e.thinkT<=0&&atkTokens()<2){
    let kind='sweep';
    if(!aliveWep&&!aliveOff)kind='lunge';
    else if(e.type==='brute'&&adx<125&&Math.random()<.55)kind='slam';
    startWind(e,kind);}
   else mv=e.strafe;}
  if(e.thinkT<=0){e.thinkT=rnd(.25,.7);e.strafe=[-1,0,0,1][(rnd(4))|0];}
  e.x+=mv*spd*dt;e.ph+=Math.abs(mv)*spd*dt*.02;}
 else if(st==='windup'){
  e.windT-=dt;
  if(!e.ticked&&e.windT<.2){e.ticked=true;SFX.tick();}
  if(e.windT<=0){e.state='attack';e.t=0;e.hitP=false;e.fired=false;e.smashed=false;
   if(e.aKind!=='shoot'&&e.aKind!=='lunge')SFX.swing(.65);}}
 else if(st==='attack'){
  const d=e.def,k=e.aKind;
  if(k==='shoot'){if(!e.fired&&e.t>=.1){e.fired=true;fireProj(e);e.recoil=1;}
   if(e.t>=d.aDur+.15){e.state='recover';e.t=0;}}
  else if(k==='melee'){
   if(e.t>=d.aA&&e.t<=d.aB&&!e.hitP){if(tryHitPlayerMelee(e,d.box,d.dmg*.5))e.hitP=true;}
   if(e.t>=d.aDur+.1){e.state='recover';e.t=0;}}
  else if(k==='lunge'){

    if(rov(b,pbox())&&vuln()){e.hitP=true;hitPlayer(8,e.dir,e);}}
   if(e.t>=.5){e.state='recover';e.t=0;}}
  else if(k==='slam'){
   if(!e.smashed&&e.t>=.16){e.smashed=true;doSlam(e);}
   if(e.t>=.55){e.state='recover';e.t=0;}}
  else{
   if(e.t>=d.aA&&e.t<=d.aB&&!e.hitP){if(tryHitPlayerMelee(e,d.box,d.dmg))e.hitP=true;}
   if(e.t>=d.aDur+.12){e.state='recover';e.t=0;}}}
 else if(st==='recover'){if(e.t>=e.def.rec){e.state='chase';e.t=0;e.thinkT=rnd(.2,.6);}}
 else if(st==='hurt'){e.x+=e.vx*dt;e.vx*=Math.exp(-dt*6);
  if(e.t>=e.hurtT){e.state='chase';e.t=0;e.thinkT=rnd(.1,.4);}}
 else if(st==='air'){
  e.vy+=1500*dt;e.y+=e.vy*dt;e.x+=e.vx*dt;e.vx*=Math.exp(-dt*1.2);e.airRot+=e.spin*dt;

   if(e.vy>420&&!e.bounced){e.bounced=true;e.vy*=-.35;e.vx*=.6;dust(e.x,GROUND,6);shake(.12);}
   else{e.vy=0;e.spin=0;e.bounced=false;e.airRot=0;e.state='chase';e.t=0;e.thinkT=rnd(.3,.6);dust(e.x,GROUND,5);}}}
 else if(st==='stagger'){e.x+=e.vx*dt;e.vx*=Math.exp(-dt*5);
  if(e.t>=e.stagT){e.state='chase';e.t=0;e.thinkT=.1;}}
 e.x=clamp(e.x,60,WORLD_W-60);}
