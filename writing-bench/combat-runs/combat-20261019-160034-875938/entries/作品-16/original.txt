function audioInit(){
  if(muted)return;
  try{
    if(!ac){
      ac=new (window.AudioContext||window.webkitAudioContext)();

      const n=ac.sampleRate*.35;noiseBuffer=ac.createBuffer(1,n,ac.sampleRate);

    }

function tone(f,d,type='sine',vol=.1,endF=f,delay=0){
  if(!ac||muted)return;
  const t=ac.currentTime+delay,o=ac.createOscillator(),g=ac.createGain();
  o.type=type;o.frequency.setValueAtTime(f,t);o.frequency.exponentialRampToValueAtTime(Math.max(20,endF),t+d);
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+d);
  o.connect(g);g.connect(master);o.start(t);o.stop(t+d+.02);
}
function noise(d=.08,vol=.12,rate=1,delay=0){
  if(!ac||muted||!noiseBuffer)return;
  const t=ac.currentTime+delay,s=ac.createBufferSource(),f=ac.createBiquadFilter(),g=ac.createGain();
  s.buffer=noiseBuffer;s.playbackRate.value=rate;f.type='bandpass';f.frequency.value=900*rate;
  g.gain.setValueAtTime(vol,t);g.gain.exponentialRampToValueAtTime(.0001,t+d);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+d+.02);
}

function enemyStrike(e,range,damage,power){
  if(e.attacked||e.dead)return false;
  const p=player,dx=p.x-e.x;
  if(Math.abs(dx)<=range&&Math.abs((p.y+38)-(e.y+38))<68){
    e.attacked=true;
    if(p.state==='parry'&&p.t>=.025&&p.t<=.205){parryEnemy(e);return true;}
    if(p.invul>0)return false;
    hurtPlayer(damage,e.facing*power);return true;
  }
  return false;
}

}

function updatePlayer(dt){
  const p=player;
  p.invul=Math.max(0,p.invul-dt);p.dashCd=Math.max(0,p.dashCd-dt);p.parryCd=Math.max(0,p.parryCd-dt);
  if(p.comboTimer>0){p.comboTimer-=dt;if(p.comboTimer<=0)p.combo=0;}
  const left=keys.KeyA||keys.ArrowLeft,right=keys.KeyD||keys.ArrowRight,move=(right?1:0)-(left?1:0);
  if(p.state==='normal'){
    if(move)p.facing=move;
    const speed=p.y>1?275:255,acc=p.y>1?850:1900;
    p.vx=approach(p.vx,move*speed,acc*dt);
  }else if(p.state==='charge'){
    p.t+=dt;p.charge=clamp(p.charge+dt/1.05,0,1);
    if(move){p.facing=move;p.vx=approach(p.vx,move*72,700*dt);}else p.vx=approach(p.vx,0,850*dt);
    if(Math.random()<dt*(4+p.charge*12))addFx('charge',p.x+rand(-25,25),p.y+rand(12,72),{max:.3,vy:80,size:1+p.charge*2,color:p.charge>.7?'#ffe58b':'#6deaff'});
  }else if(p.state==='attack'){
    const a=p.attack;a.t+=dt;p.t=a.t;
    attackHits();
    if(a.kind==='air')p.vy=Math.max(p.vy,-210);
    p.vx=approach(p.vx,0,(a.kind==='heavy'?550:390)*dt);
    if(a.t>=a.dur){
      if(a.kind==='light'&&p.queued&&a.stage<3)startAttack('light',a.stage+1);
      else{p.state='normal';p.attack=null;p.queued=false;}
    }
  }else if(p.state==='dash'){
    p.t-=dt;p.vx=p.facing*760;
    if(p.t<=0){p.state='normal';p.vx=p.facing*230;}
  }else if(p.state==='parry'){
    p.t+=dt;p.vx=approach(p.vx,0,1500*dt);
    if(p.t>.43)p.state='normal';
  }else if(p.state==='counter'){
    p.t-=dt;p.vx=0;
    if(p.t<=0)p.state='normal';
  }else if(p.state==='hurt'){
    p.t-=dt;p.vx=approach(p.vx,0,420*dt);
    if(p.t<=0)p.state='normal';
  }else if(p.state==='dead'){
    p.t-=dt;p.vx=approach(p.vx,0,280*dt);
  }
  if(p.state!=='dash'){
    p.vy-=1750*dt;
  }
  const oldY=p.y;p.x+=p.vx*dt;p.y+=p.vy*dt;p.x=clamp(p.x,55,WORLD-55);
  if(p.y<=0){
    if(oldY>10&&p.vy<-250){addDust(p.x,0,Math.min(12,4+Math.floor(-p.vy/100)),'#7bd6df');shake=Math.max(shake,Math.min(4,-p.vy/180));}
    p.y=0;p.vy=0;p.grounded=true;p.airAttack=false;
  }else p.grounded=false;
  p.run+=Math.abs(p.vx)*dt*.035;
}
function updateEnemy(e,dt){
  if(e.flash>0)e.flash-=dt;
  if(e.dead){
    e.deathT-=dt;e.vy-=1500*dt;e.x+=e.vx*dt;e.y+=e.vy*dt;e.vx*=Math.pow(.1,dt);
    if(e.y<0){e.y=0;e.vy=0;}return;
  }
  e.cooldown-=dt;
  if(e.state==='hurt'){
    e.t-=dt;
    if(e.t<=0)e.state='idle';
  }else if(e.type==='ronin')updateRonin(e,dt);
  else updateMantis(e,dt);
  e.vy-=1600*dt;e.x+=e.vx*dt;e.y+=e.vy*dt;
  if(e.y<=0){if(e.y<0&&e.vy<-260)addDust(e.x,0,5,'#b45a51');e.y=0;e.vy=0;}
  e.x=clamp(e.x,55,WORLD-55);
}
function updateRonin(e,dt){
  const d=player.x-e.x,ad=Math.abs(d),sword=limbAlive(e,'sword'),arms=aliveLimbCount(e);
  if(e.state==='idle'){
    if(ad>12)e.facing=d>0?1:-1;
    if(e.y>2){e.vx*=.98;return;}
    if(e.cooldown<=0&&ad<(sword?125:78)){
      e.state='windup';e.t=sword?(arms===2?.46:.61):.57;e.vx=0;e.attackType=sword?'slash':'kick';e.attacked=false;
    }else{
      const desired=ad>(sword?82:58)?e.facing*(arms?112:82):0;
      e.vx=approach(e.vx,desired,650*dt);
    }
  }else if(e.state==='windup'){
    e.t-=dt;e.vx=approach(e.vx,0,900*dt);
    if(e.t<=0){e.state='attack';e.t=e.attackType==='slash'?.18:.2;e.attacked=false;SFX.swing(.75);}
  }else if(e.state==='attack'){
    if(!e.attacked)enemyStrike(e,e.attackType==='slash'?91:59,e.attackType==='slash'?(aliveLimbCount(e)>1?14:10):8,e.attackType==='slash'?330:240);
    e.vx=e.facing*(e.attackType==='slash'?170:125);e.t-=dt;
    if(e.t<=0){e.state='recover';e.t=e.attackType==='slash'?.38:.45;e.cooldown=rand(.55,.9);e.vx*=.2;}
  }else if(e.state==='recover'){
    e.t-=dt;e.vx=approach(e.vx,0,700*dt);if(e.t<=0)e.state='idle';
  }
}
function updateMantis(e,dt){
  const d=player.x-e.x,ad=Math.abs(d),blades=aliveLimbCount(e);
  if(e.state==='idle'){
    e.facing=d>0?1:-1;
    if(e.y>2){e.vx*=.99;return;}
    if(e.cooldown<=0&&ad<390){
      e.state='windup';e.t=blades?.58:.75;e.vx=0;e.attacked=false;e.attackType=blades?'pounce':'ram';
    }else{
      let desired=0;if(ad>245)desired=e.facing*145;else if(ad<135)desired=-e.facing*90;
      e.vx=approach(e.vx,desired,780*dt);
    }
  }else if(e.state==='windup'){
    e.t-=dt;e.vx=approach(e.vx,0,1100*dt);
    if(e.t<=0){
      e.state='leap';e.t=blades?.48:.42;e.attacked=false;e.vx=e.facing*(blades?530:390);e.vy=blades?230:120;
      SFX.dash();
    }
  }else if(e.state==='leap'){
    if(!e.attacked)enemyStrike(e,blades?64:50,blades?12+blades*2:8,blades?390:270);
    e.t-=dt;if(e.t<=0){e.state='recover';e.t=.5;e.cooldown=rand(.75,1.15);e.vx*=.25;}
  }else if(e.state==='recover'){
    e.t-=dt;e.vx=approach(e.vx,0,800*dt);if(e.t<=0)e.state='idle';
  }
}
function updateParts(dt){
  for(const p of parts){
    p.life-=dt;p.vy-=1300*dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.rot+=p.vr*dt;p.vx*=Math.pow(.28,dt);
    if(p.y<0){p.y=0;if(Math.abs(p.vy)>90){p.vy=-p.vy*.3;p.vr*=.7;addDust(p.x,0,2,'#a85b50');}else p.vy=0;}
  }
  parts=parts.filter(p=>p.life>0);
}
function updateFx(dt){
  for(const f of fx){f.t+=dt;f.x+=f.vx*dt;f.y+=f.vy*dt;if(f.type==='spark'||f.type==='dust')f.vy-=520*dt;}
  fx=fx.filter(f=>f.t<f.max);
}
function update(dt){
  clock+=dt;
  if(!playing)return;
  updatePlayer(dt);
  if(playerAlive)for(const e of enemies)updateEnemy(e,dt);
  updateParts(dt);updateFx(dt);
  enemies=enemies.filter(e=>!e.dead||e.deathT>0);
  if(playerAlive){
    spawnTimer-=dt;
    const cap=Math.min(5,2+Math.floor(kills/4));
    if(livingEnemies()<cap&&spawnTimer<=0){
      const side=(Math.random()<.5?-1:1);
      spawn((kills+livingEnemies()+Math.floor(clock))%3===0?'mantis':'ronin',side);
      spawnTimer=rand(.55,1.1);
    }
  }
  const target=clamp(player.x-W*.5+player.facing*55,0,Math.max(0,WORLD-W));
  camera=lerp(camera,target,1-Math.pow(.0008,dt));
  shake=Math.max(0,shake-dt*18);
}

      ctx.lineWidth=(f.big?12:7)*a;ctx.shadowColor=f.color;ctx.shadowBlur=14*a;
      ctx.beginPath();ctx.arc(-12,5,(f.big?78:60)*f.size,-1.75,.8);ctx.stroke();
      ctx.lineWidth=2;ctx.strokeStyle='#fff';ctx.stroke();
    }else if(f.type==='ring'||f.type==='parry'){
      ctx.strokeStyle=f.color;ctx.lineWidth=(f.type==='parry'?6:4)*a;ctx.beginPath();ctx.arc(x,y,(8+u*42)*f.size,0,7);ctx.stroke();
      if(f.type==='parry'){ctx.beginPath();ctx.arc(x,y,(25+u*70),-.6,.5);ctx.stroke();}
    }else if(f.type==='break'){
      ctx.strokeStyle=f.color;ctx.lineWidth=5*a;
      for(let i=0;i<6;i++){const an=i*1.047+u;ctx.beginPath();ctx.moveTo(x+Math.cos(an)*10,y+Math.sin(an)*10);ctx.lineTo(x+Math.cos(an)*(35+u*38),y+Math.sin(an)*(35+u*38));ctx.stroke();}
    }else if(f.type==='charge'){
      ctx.fillStyle=f.color;ctx.beginPath();ctx.arc(x,y,2.5*f.size*a,0,7);ctx.fill();
    }else if(f.type==='portal'){
      ctx.strokeStyle=f.color;ctx.lineWidth=4*a;ctx.beginPath();ctx.ellipse(x,y-36,18+u*25,55*(1-u*.3),0,0,7);ctx.stroke();
    }else if(f.type==='death'){
      ctx.strokeStyle=f.color;ctx.lineWidth=8*a;ctx.beginPath();ctx.arc(x,y,18+u*75,0,7);ctx.stroke();
    }
    ctx.restore();
  }
}
