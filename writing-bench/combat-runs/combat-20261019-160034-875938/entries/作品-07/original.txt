let ac=null,muted=false;
function initAudio(){
  if(muted)return;
  try{
    if(!ac)ac=new (window.AudioContext||window.webkitAudioContext)();
    if(ac.state==="suspended")ac.resume();
  }catch(_){}
}
function osc(freq,dur,gain,type="sine",end=freq*.55,delay=0){
  if(!ac||muted)return;
  const t=ac.currentTime+delay,o=ac.createOscillator(),g=ac.createGain();
  o.type=type;o.frequency.setValueAtTime(freq,t);o.frequency.exponentialRampToValueAtTime(Math.max(20,end),t+dur);
  g.gain.setValueAtTime(.0001,t);g.gain.exponentialRampToValueAtTime(gain,t+.008);g.gain.exponentialRampToValueAtTime(.0001,t+dur);
  o.connect(g).connect(ac.destination);o.start(t);o.stop(t+dur+.02);
}
function noise(dur=.08,gain=.08,cut=1200){
  if(!ac||muted)return;
  const n=Math.max(1,Math.floor(ac.sampleRate*dur)),b=ac.createBuffer(1,n,ac.sampleRate),d=b.getChannelData(0);
  for(let i=0;i<n;i++)d[i]=(Math.random()*2-1)*(1-i/n);
  const s=ac.createBufferSource(),f=ac.createBiquadFilter(),g=ac.createGain();
  s.buffer=b;f.type="bandpass";f.frequency.value=cut;f.Q.value=.7;g.gain.value=gain;
  s.connect(f).connect(g).connect(ac.destination);s.start();
}

function updatePlayer(dt){
  const p=player;if(!p.alive)return;
  p.inv=Math.max(0,p.inv-dt);p.dashCd=Math.max(0,p.dashCd-dt);
  p.comboTimer-=dt;if(p.comboTimer<=0)p.combo=0;

    p.hurt-=dt;p.vx*=Math.pow(.03,dt);

    if(pressed.KeyL&&(!p.action||p.action.type!=="dash"))beginDash();

    }
    if(p.action)processPlayerAction(dt);
    let scale=1;
    if(p.action){
      if(p.action.type==="charge")scale=.28;
      else if(["quick","heavy","special"].includes(p.action.type))scale=.42;
      else if(p.action.type==="plunge")scale=.2;

function processPlayerAction(dt){
  const p=player,a=p.action;if(!a)return;
  a.t+=dt;
  if(a.type==="dash"){
    if(a.t>=a.dur)p.action=null;
    return;
  }
  if(a.type==="charge"){
    p.chargeGlow=Math.min(1,a.t/.9);
    if(!keys.KeyK||a.t>=1.05)startHeavy(a.t/1.0);
    return;
  }
  if(a.type==="plunge"){
    if(a.t>.12)p.vy=Math.max(p.vy,920);
    if(a.t>.8&&!p.grounded)p.action=null;
    return;
  }
  if(a.type==="quick"){
    const defs=[
      {on:.075,off:.16,range:91,damage:13,knock:165,limb:5,stop:.035},
      {on:.065,off:.17,range:107,damage:15,knock:205,limb:7,stop:.046},
      {on:.12,off:.245,range:128,damage:24,knock:430,limb:15,stop:.078}
    ],d=defs[a.n-1];
    if(pressed.KeyJ&&a.t>.065)a.queued="J";
    if(pressed.KeyK&&a.t>.11)a.queued="K";

      if(!a.sounded){a.sounded=true;sfx("swing",a.n===3?1.4:1);addSlash(p.x+p.face*30,p.y-48,p.face,65+a.n*13,a.n===3?"#fff0b2":"#81f5ff")}
      strikeEnemies(a,{...d,launch:a.n===3?130:0,kind:"quick"});
    }
    if(a.t>=a.dur){
      const q=a.queued,n=a.n;p.action=null;
      if(q==="J")startQuick(n===3?1:n+1);
      else if(q==="K")startCharge();
    }
    return;
  }
  if(a.type==="heavy"){
    const on=.17,off=.285,c=a.charge;
    if(a.t>=on&&a.t<=off){
      if(!a.sounded){a.sounded=true;sfx("swing",1.8);addSlash(p.x+p.face*40,p.y-48,p.face,105+c*38,"#ffd675")}

    }
    if(a.t>=a.dur)p.action=null;
  }
}

function strikeEnemies(action,d){

  for(const e of enemies){
    if(e.dead||action.hits.has(e.id))continue;

}

  p.hp-=dmg;p.hurt=.32;p.inv=.72;p.action=null;p.vx=dir*360;p.vy=-180;

  if(p.hp<=0){p.hp=0;p.alive=false;game.playing=false;p.vy=-330;game.shake=13}

}

function updateEnemies(dt){
  for(const e of enemies){
    if(e.dead){e.deadT+=dt;e.vy+=1300*dt;e.x+=e.vx*dt;e.y+=e.vy*dt;if(e.y>GROUND){e.y=GROUND;e.vy=0;e.vx*=.86}continue}
    e.flash=Math.max(0,e.flash-dt);e.face=player.x<e.x?-1:1;

      e.hitstun-=dt;e.vx=approach(e.vx,0,650*dt);
    }else if(e.mode==="spawn"){
      e.t+=dt;e.vx=0;if(e.t>.42){e.mode="chase";e.t=0}
    }else if(e.type==="blade")updateBlade(e,dt);
    else updateGunner(e,dt);

    e.vy+=1350*dt;e.x+=e.vx*dt;e.y+=e.vy*dt;
    e.x=Math.max(-1120,Math.min(1120,e.x));
    if(e.y>=GROUND){
      if(!e.grounded&&e.vy>300)dust(e.x,e.y,5);
      e.y=GROUND;e.vy=0;e.grounded=true;
    }else e.grounded=false;
  }
  enemies=enemies.filter(e=>!e.dead||e.deadT<1.15);
}
function updateBlade(e,dt){
  const dist=Math.abs(player.x-e.x),speed=e.limbs.leg.on?145:64;
  if(e.mode==="chase"){
    if(dist>(e.limbs.weapon.on?93:66))e.vx=e.face*speed;else e.vx=approach(e.vx,0,1100*dt);
    if(dist<(e.limbs.weapon.on?122:79)){e.mode="attack";e.t=0;e.did=false;e.vx=0}
  }else if(e.mode==="attack"){
    e.t+=dt;const armed=e.limbs.weapon.on,active=armed?.43:.58,end=armed?.82:.94;
    if(e.t>active-.055&&e.t<active&&!e.did)sfx("swing",.75);
    if(e.t>=active&&!e.did){
      e.did=true;e.vx=e.face*(armed?290:180);
      if(Math.abs(player.x-e.x)<(armed?122:76)&&Math.abs(player.y-e.y)<90){
        if(isParrying())triggerParry(e);else hurtPlayer(armed?15:9,e.face);
      }
    }
    if(e.t>end){e.mode="chase";e.t=0;e.did=false}
  }
}
function updateGunner(e,dt){
  const dist=Math.abs(player.x-e.x),speed=e.limbs.leg.on?108:47;
  if(!e.limbs.weapon.on){
    if(e.mode==="attack"&&e.t<.2){e.mode="chase";e.t=0}
    if(e.mode==="chase"){
      if(dist>72)e.vx=e.face*speed;else{e.vx=0;e.mode="ram";e.t=0;e.did=false}
    }else if(e.mode==="ram"){
      e.t+=dt;
      if(e.t>.55&&!e.did){e.did=true;e.vx=e.face*260;if(dist<85){if(isParrying())triggerParry(e);else hurtPlayer(11,e.face)}}
      if(e.t>.9){e.mode="chase";e.t=0}
    }
    return;
  }
  if(e.mode==="chase"){
    if(dist<210)e.vx=-e.face*speed;
    else if(dist>410)e.vx=e.face*speed;
    else e.vx=approach(e.vx,0,900*dt);
    e.t+=dt;
    if(e.t>1.05&&dist<540){e.mode="attack";e.t=0;e.shot=false;e.aimX=player.x;e.aimY=player.y-43;e.vx=0}
  }else if(e.mode==="attack"){
    e.t+=dt;e.vx=0;
    if(e.t<.5){e.aimX=player.x;e.aimY=player.y-43}
    if(e.t>=.72&&!e.shot){
      e.shot=true;const sx=e.x+e.face*34,sy=e.y-51,dx=e.aimX-sx,dy=e.aimY-sy,l=Math.hypot(dx,dy)||1;
      projectiles.push({x:sx,y:sy,vx:dx/l*590,vy:dy/l*590,life:1.4,owner:e});sfx("shot");
    }
    if(e.t>.98){e.mode="chase";e.t=0}
  }
}
function updateProjectiles(dt){
  for(const b of projectiles){
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(!player.alive)continue;
    if(Math.abs(b.x-player.x)<23&&b.y>player.y-78&&b.y<player.y+5){
      if(isParrying()){triggerParry(b.owner);b.life=0}
      else if(isDashing()){spawnBurst(b.x,b.y,"#66efff",5,180);b.life=0}
      else if(hurtPlayer(13,Math.sign(b.vx))){b.life=0}
    }
  }
  projectiles=projectiles.filter(b=>b.life>0&&Math.abs(b.x-player.x)<1200);
}
function spawnManager(dt){
  const desired=Math.min(3,2+Math.floor((game.wave-1)/3)),live=aliveEnemies();
  if(live<desired){
    game.spawnTimer-=dt;
    if(game.spawnTimer<=0){
      let side=Math.random()<.5?-1:1,x=player.x+side*(350+Math.random()*150);
      if(x<-1050||x>1050){side*=-1;x=player.x+side*(350+Math.random()*120)}
      spawnEnemy((game.kills+live)%2?"blade":"gun",Math.max(-1050,Math.min(1050,x)));
      game.spawnTimer=.8;
    }
  }else game.spawnTimer=.7;
}
function approach(v,t,a){return v<t?Math.min(t,v+a):Math.max(t,v-a)}

function dust(x,y,n){

function updateFX(dt){

  particles=particles.filter(p=>p.life>0);
  for(const s of slashes)s.life-=dt;slashes=slashes.filter(s=>s.life>0);
  for(const t of texts){t.life-=dt;t.y-=35*dt}texts=texts.filter(t=>t.life>0);
  for(const d of debris){
    d.life-=dt;d.vy+=1200*dt;d.x+=d.vx*dt;d.y+=d.vy*dt;d.rot+=d.vr*dt;

  }
  debris=debris.filter(d=>d.life>0);
  game.shake=Math.max(0,game.shake-38*dt);
}
