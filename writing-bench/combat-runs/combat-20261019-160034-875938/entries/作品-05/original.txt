var AC=null, MAS=null, NOISE=null, muted=false;

function ensureAudio(){
  if(AC){ if(AC.state === 'suspended'){ try{ AC.resume(); }catch(e){} } return; }
  try{
    var C = window.AudioContext || window.webkitAudioContext;
    if(!C) return;
    AC = new C();
    MAS = AC.createGain();
    MAS.gain.value = muted ? 0 : 0.5;
    MAS.connect(AC.destination);
    var n = Math.floor(AC.sampleRate * 0.4);
    NOISE = AC.createBuffer(1, n, AC.sampleRate);
    var d = NOISE.getChannelData(0);
    for(var i=0;i<n;i++) d[i] = Math.random()*2-1;
  }catch(e){ AC = null; }
}
function tone(f0,f1,dur,vol,type){
  if(!AC || muted) return;
  try{
    var t = AC.currentTime;
    var o = AC.createOscillator(), g = AC.createGain();
    o.type = type||'sine';
    o.frequency.setValueAtTime(f0, t);
    o.frequency.exponentialRampToValueAtTime(Math.max(20,f1), t+dur);
    g.gain.setValueAtTime(0.0001, t);
    g.gain.exponentialRampToValueAtTime(Math.max(0.0002,vol), t+0.008);
    g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
    o.connect(g); g.connect(MAS);
    o.start(t); o.stop(t+dur+0.03);
  }catch(e){}
}
function noiseHit(dur,f0,f1,vol,q){
  if(!AC || muted || !NOISE) return;
  try{
    var t = AC.currentTime;
    var s = AC.createBufferSource(); s.buffer = NOISE;
    var f = AC.createBiquadFilter(); f.type='bandpass'; f.Q.value = q||1.2;
    f.frequency.setValueAtTime(f0, t);
    f.frequency.exponentialRampToValueAtTime(Math.max(60,f1), t+dur);
    var g = AC.createGain();
    g.gain.setValueAtTime(vol, t);
    g.gain.exponentialRampToValueAtTime(0.0001, t+dur);
    s.connect(f); f.connect(g); g.connect(MAS);
    s.start(t); s.stop(t+dur+0.03);
  }catch(e){}
}

function setState(s){ P.state = s; P.t = 0; }

function startAttack(id){
  var a = ATK[id];
  P.cur = a; P.curId = id;
  P.state = (id === 'heavy') ? 'heavy' : 'attack';
  P.t = 0; P.phase = 'startup'; P.pt = 0;
  P.hitSet = new Set();
  P.dmg = a.dmg; P.lb = a.lb;
  if(id === 'heavy'){ P.dmg = a.dmg + P.power*0.6; P.lb = a.lb + P.power*0.7; }
  if(id === 'plunge'){ P.vy = Math.max(P.vy, 12); }
  if(id === 'dashA'){ P.vx = P.face*7; }
  if(!P.onGround && id === 'a1'){ P.cur = ATK.air; P.curId = 'air'; }
}

function startDash(){
  P.state = 'dash'; P.t = 0;
  P.vx = P.face * 13.5;
  P.vy = Math.min(P.vy, 0);
  P.dashCd = 20; P.invuln = Math.max(P.invuln, 11);
  SFX.dash();
  for(var i=0;i<6;i++) addFx({t:'spark', x:P.x-P.face*rnd(6,26), y:P.y-rnd(20,80), vx:-P.face*rnd(1,6), vy:rnd(-2,2),
    life:rnd(10,20), max:20, color:'#7fe4ff', size:rnd(1.5,3)});
  G.shake = Math.max(G.shake, 2.5);
}

function startParry(){

}

function doJump(){

  } else if(P.airJump > 0){
    P.airJump--; P.vy = -10.4; P.airAtk = 2;
    SFX.dash();
    addFx({t:'ring', x:P.x, y:P.y-8, r:8, vr:3.4, life:16, max:16, color:'rgba(150,220,255,1)', w:3});
  }
}

function canAct(){
  var s = P.state;
  if(s === 'idle' || s === 'run' || s === 'jump' || s === 'fall' || s === 'dash') return true;
  if((s === 'attack' || s === 'heavy') && P.phase === 'recovery' && P.pt >= 2) return true;
  return false;
}

function updatePlayer(){
  var p = P;
  p.t++;
  if(p.invuln > 0) p.invuln--;
  if(p.dashCd > 0) p.dashCd--;
  if(p.parryCd > 0) p.parryCd--;
  if(G.comboT > 0){ G.comboT--; if(G.comboT === 0) G.combo = 0; }

  var dir = (keys['d']||keys['arrowright']?1:0) - (keys['a']||keys['arrowleft']?1:0);

  /* ---- 动作触发 ---- */
  if(!p.dead && canAct()){
    if(buf.parry > 0 && p.parryCd <= 0 && p.state !== 'dash'){ buf.parry = 0; startParry(); }
    else if(buf.dash > 0 && p.dashCd <= 0 && p.state !== 'dash'){ buf.dash = 0; startDash(); }
    else if(buf.light > 0){ buf.light = 0; doLight(); }
    else if(buf.heavy > 0 && !p.onGround && p.state !== 'dash'){ buf.heavy = 0; startAttack('plunge'); }
    else if(buf.jump > 0 && p.state !== 'dash'){ buf.jump = 0; doJump(); }
    else if(keys['k'] && p.onGround && p.state !== 'dash' && p.state !== 'run'){ }
    else if(keys['k'] && p.onGround && p.state !== 'dash'){ p.state='charge'; p.t=0; p.charge=0; p.vx=0; }
  }

  /* ---- 状态行为 ---- */
  switch(p.state){
    case 'idle': case 'run': case 'jump': case 'fall': {
      var spd = p.onGround ? 5.7 : 4.6;
      p.vx = lerp(p.vx, dir*spd, p.onGround?0.34:0.16);
      if(dir) p.face = dir;
      if(p.onGround) p.state = Math.abs(p.vx) > 0.7 ? 'run' : 'idle';
      else p.state = p.vy < 0 ? 'jump' : 'fall';
      break;
    }
    case 'charge': {
      p.charge = Math.min(36, p.charge + 1);
      p.vx = lerp(p.vx, dir*1.1, 0.2);
      if(dir) p.face = dir;
      if(!keys['k']){
        p.power = p.charge;
        startAttack('heavy');
        SFX.heavySwing();
      }
      if(p.charge % 9 === 0) addFx({t:'ring', x:p.x, y:p.y-60, r:20+p.charge*0.9, vr:1.6, life:14, max:14,
        color:'rgba(255,190,90,0.85)', w:2});
      break;
    }
    case 'dash': {
      p.vx = lerp(p.vx, 0, 0.12);
      if(p.t > 13){ setState(p.onGround?'idle':'fall'); }
      if(p.t % 3 === 0) G.after.push({x:p.x, y:p.y, face:p.face, life:12, max:12, pose:getPose()});
      break;
    }
    case 'parry': {
      p.vx *= 0.82;
      if(p.t >= 24) setState(p.onGround?'idle':'fall');
      break;
    }
    case 'attack': case 'heavy': {
      var a = p.cur;
      p.pt++;
      if(p.phase === 'startup'){
        p.vx = lerp(p.vx, p.face*(p.curId==='heavy'?1.4:2.6), 0.35);
        if(p.pt >= a.s){ p.phase = 'active'; p.pt = 0; onActive(); }
      } else if(p.phase === 'active'){
        p.vx = lerp(p.vx, p.face*(p.curId==='heavy'?0.9:1.8), 0.3);
        if(p.pt >= a.a){ p.phase = 'recovery'; p.pt = 0; }
      } else {
        p.vx *= 0.78;
        if(buf.light > 0 && a.next && p.pt >= 2){ buf.light = 0; startAttack(a.next); break; }
        if(p.pt >= a.r){ setState(p.onGround?'idle':'fall'); }
      }
      break;
    }
    case 'hurt': {
      p.vx *= 0.9;
      if(p.t >= 20) setState(p.onGround?'idle':'fall');
      break;
    }
  }

  /* ---- 物理 ---- */
  p.vy = Math.min(p.vy + 0.74, 23);
  p.x += p.vx;
  p.y += p.vy;
  if(p.y >= GROUND){
    if(p.vy > 9 && p.state !== 'charge'){
      puff(p.x, GROUND, 8, '#8ea4d8');
      G.shake = Math.max(G.shake, 2);
    }
    p.y = GROUND;
    if(p.vy > 2 && (p.state === 'plunge')){ /* handled in active */ }
    p.vy = 0; p.onGround = true;
    p.airJump = 1; p.airAtk = 2;
    if(p.state === 'jump' || p.state === 'fall') p.state = 'idle';
  } else {
    p.onGround = false;
  }
  p.x = clamp(p.x, 40, ARENA_W-40);

  /* ---- 命中判定 ---- */
  if((p.state === 'attack' || p.state === 'heavy') && p.phase === 'active') resolveAttack();
}

function doLight(){

  if((P.state === 'attack' || P.state === 'heavy') && P.cur && P.cur.next){ startAttack(P.cur.next); return; }
  if(!P.onGround){
    if(P.airAtk <= 0) return;
    P.airAtk--;
    startAttack('air');
    return;
  }
  startAttack('a1');
}

  G.arcs.push(ace);
  if(P.curId === 'plunge'){

  }
}

function attackBox(){
  var a = P.cur, d = P.face;
  return { x: d>0 ? P.x-8 : P.x-a.reach+8, y: P.y-a.top, w: a.reach, h: a.top-a.bot };
}
function circRect(cx,cy,r,b){
  var nx = clamp(cx, b.x, b.x+b.w), ny = clamp(cy, b.y, b.y+b.h);
  var dx = cx-nx, dy = cy-ny;
  return dx*dx+dy*dy <= r*r;
}

function resolveAttack(){
  var box = attackBox();
  var i, e;
  for(i=0;i<G.enemies.length;i++){
    e = G.enemies[i];
    if(e.dead || P.hitSet.has(e)) continue;
    if(!circRect(e.x, e.y-e.h*0.5, Math.max(e.w*0.6, e.h*0.42), box)) continue;
    var target = null, bd = 1e9, j, L;
    for(j=0;j<e.limbs.length;j++){
      L = e.limbs[j];
      if(L.broken) continue;
      if(circRect(L.x, L.y, L.r+3, box)){
        var dd = Math.hypot(L.x-P.x, L.y-P.y);
        if(dd < bd){ bd = dd; target = L; }
      }
    }
    P.hitSet.add(e);
    hitEnemy(e, target, box);
  }
  // 击碎弹幕
  for(i=0;i<G.proj.length;i++){
    var pr = G.proj[i];
    if(!pr.dead && circRect(pr.x, pr.y, pr.r+2, box)){
      pr.dead = true;
      burst(pr.x, pr.y, 8, '#7fe4ff');
      SFX.hit(0.3);
    }
  }
}

function hitEnemy(e, limb, box){
  var a = P.cur;
  var dmg = P.dmg;
  e.hp -= dmg;
  e.hitFlash = 7;
  e.stun = Math.max(e.stun, a.stun);
  var dir = sign(e.x - P.x) || P.face;
  e.vx += dir * a.kb;
  if(a.launch) e.vy = a.launch;

  var hx = limb ? limb.x : e.x;
  var hy = limb ? limb.y : e.y - e.h*0.6;
  hx = clamp(hx, box.x-6, box.x+box.w+6);

  G.hitCount++;
  G.combo++; G.comboT = 78;
  if(G.combo > G.best) G.best = G.combo;

  var power = clamp(dmg/26, 0.15, 1);
  G.hitstop = Math.max(G.hitstop, a.hs);
  G.shake = Math.max(G.shake, 2.2 + a.hs*0.55);
  SFX.hit(power);

  burst(hx, hy, Math.round(6 + a.hs), power>0.55 ? '#ffd08a' : '#bfe9ff');
  addFx({t:'flash', x:hx, y:hy, r:14+a.hs*1.9, life:6, max:6, color:power>0.55?'rgba(255,205,130,':'rgba(170,235,255,'});
  G.arcs.push({x:hx, y:hy, face:1, r:6, a0:0, a1:6.283, w:4, life:8, max:8, color:'rgba(255,255,255,0.7)'});

  if(limb){
    limb.hp -= P.lb;
    if(limb.hp <= 0 && !limb.broken) breakLimb(e, limb);
  }
  if(e.hp <= 0) killEnemy(e);
}

function breakLimb(e, L){
  L.broken = true;
  G.limbBreakCount++;
  G.hitstop = Math.max(G.hitstop, 12);
  G.shake = Math.max(G.shake, 9);
  SFX.brk();
  var i;
  for(i=0;i<12;i++){
    addFx({t:'shard', x:L.x, y:L.y, vx:rnd(-7,7)+sign(e.x-P.x)*3, vy:rnd(-11,-2),
      rot:rnd(0,6.28), vr:rnd(-0.4,0.4), life:rnd(40,75), max:75,
      size:rnd(4,11), color: L.core ? '#e8d2a8' : '#c09064'});
  }
  for(i=0;i<10;i++){
    addFx({t:'spark', x:L.x, y:L.y, vx:rnd(-9,9), vy:rnd(-9,3), life:rnd(10,22), max:22,
      color:'#ffd08a', size:rnd(1.5,3.5)});
  }
  addFx({t:'ring', x:L.x, y:L.y, r:8, vr:6, life:20, max:20, color:'rgba(255,215,150,1)', w:4});
  if(L.core){
    killEnemy(e, true);
  } else {
    e.stun = Math.max(e.stun, 26);
  }
}

function killEnemy(e, byHead){
  if(e.dead) return;
  e.dead = true;
  G.hitstop = Math.max(G.hitstop, 10);
  G.shake = Math.max(G.shake, 11);
  SFX.die();
  var i, c = byHead ? 34 : 22;
  for(i=0;i<c;i++){
    addFx({t:'shard', x:e.x+rnd(-14,14), y:e.y-rnd(20,e.h), vx:rnd(-9,9), vy:rnd(-13,-2),
      rot:rnd(0,6.28), vr:rnd(-0.4,0.4), life:rnd(45,85), max:85, size:rnd(4,13), color:'#b98b5e'});
  }
  burst(e.x, e.y-e.h*0.55, 20, '#ffd6a0');
  addFx({t:'ring', x:e.x, y:e.y-e.h*0.5, r:10, vr:7, life:22, max:22, color:'rgba(255,210,150,1)', w:5});
  P.hp = Math.min(P.maxHp, P.hp + 4);
  var idx = G.enemies.indexOf(e);
  if(idx >= 0) G.enemies.splice(idx, 1);
  G.haz = G.haz.filter(function(h){ return h.owner !== e; });
}

function hurtPlayer(dmg, dir, src){
  if(P.dead || P.invuln > 0) return false;
  if(P.state === 'dash') return false;
  // 见切
  if(P.state === 'parry' && P.t >= 3 && P.t <= 16){
    doParry(src);
    return true;
  }
  P.hp -= dmg;
  P.invuln = 56;
  P.state = 'hurt'; P.t = 0;
  P.vx = dir * 6; P.vy = -4.2;
  P.combo = 0;
  G.combo = 0; G.comboT = 0;
  G.shake = Math.max(G.shake, 9);
  G.hurtFlash = 14;
  SFX.hurt();
  burst(P.x, P.y-50, 10, '#ff6f8a');
  if(P.hp <= 0){
    P.hp = 0; P.dead = true;
    G.state = 'dead';
    G.hitstop = 14;
    burst(P.x, P.y-45, 34, '#ff5f7f');
  }
  return true;
}

function doParry(src){
  G.hitstop = Math.max(G.hitstop, 16);
  G.shake = Math.max(G.shake, 7);
  SFX.parry();
  P.parryCd = 0;
  P.state = 'idle'; P.t = 0;
  P.invuln = Math.max(P.invuln, 12);
  addFx({t:'ring', x:P.x+P.face*26, y:P.y-62, r:6, vr:8, life:20, max:20, color:'rgba(190,240,255,1)', w:5});
  addFx({t:'flash', x:P.x+P.face*26, y:P.y-62, r:48, life:9, max:9, color:'rgba(190,240,255,'});
  burst(P.x+P.face*26, P.y-62, 16, '#cfeeff');
  G.combo++; G.comboT = 90;
  if(src && !src.dead){
    if(src.hp !== undefined){
      src.stun = 72;
      src.hitFlash = 10;
      src.hp -= 16;
      G.hitCount++;
      var L = src.limbs.find(function(l){ return !l.broken && !l.core && (l.id==='armR'||l.id==='cannon'); });
      if(L){ L.hp -= 30; if(L.hp <= 0) breakLimb(src, L); }
      if(src.hp <= 0) killEnemy(src);
    }
  }
  // 清掉附近弹幕
  G.proj.forEach(function(pr){
    if(!pr.dead && Math.hypot(pr.x-P.x, pr.y-(P.y-60)) < 150){ pr.dead = true; burst(pr.x,pr.y,8,'#9fe8ff'); }
  });
}

/* ============================== 特效 ============================== */
function addFx(o){ if(G.fx.length < 620) G.fx.push(o); }
function burst(x,y,n,color){
  for(var i=0;i<n;i++){
    var a = rnd(0,6.283), s = rnd(1.5,9);
    addFx({t:'spark', x:x, y:y, vx:Math.cos(a)*s, vy:Math.sin(a)*s-2,
      life:rnd(9,22), max:22, color:color, size:rnd(1.4,3.4)});
  }
}
function puff(x,y,n,color){
  for(var i=0;i<n;i++){
    addFx({t:'spark', x:x+rnd(-10,10), y:y-rnd(0,6), vx:rnd(-3.5,3.5), vy:rnd(-3,-0.4),
      life:rnd(10,20), max:20, color:color, size:rnd(2,4.5)});
  }
}

function updateFx(k){
  var i, f;
  for(i=G.fx.length-1;i>=0;i--){
    f = G.fx[i];
    f.life -= k;
    if(f.life <= 0){ G.fx.splice(i,1); continue; }
    if(f.t === 'spark'){
      f.x += f.vx*k; f.y += f.vy*k;
      f.vy += 0.42*k; f.vx *= 0.97;
    } else if(f.t === 'shard'){
      f.x += f.vx*k; f.y += f.vy*k;
      f.vy += 0.55*k; f.vx *= 0.985; f.rot += f.vr*k;
      if(f.y > GROUND){ f.y = GROUND; f.vy *= -0.36; f.vx *= 0.6; f.vr *= 0.6; }
    } else if(f.t === 'ring'){
      f.r += f.vr*k; f.vr *= 0.94;
    }
  }
  for(i=G.arcs.length-1;i>=0;i--){
    G.arcs[i].life -= k;
    if(G.arcs[i].life <= 0) G.arcs.splice(i,1);
  }
  for(i=G.after.length-1;i>=0;i--){
    G.after[i].life -= k;
    if(G.after[i].life <= 0) G.after.splice(i,1);
  }
}

            {id:'cannon', ox:20, oy:-64, r:13, hp:15},

function spawnEnemy(type, x){
  var d = ETYPES[type];
  var hpMul = 1 + (G.wave-1)*0.13;
  var e = {
    type:type, x:clamp(x, 60, ARENA_W-60), y:GROUND, vx:0, vy:0, face:1,
    hp:Math.round(d.hp*hpMul), maxHp:Math.round(d.hp*hpMul),
    w:d.w, h:d.h, speed:d.speed*(1+(G.wave-1)*0.04),
    st:'idle', t:0, stun:0, hitFlash:0, atkCd:40, dead:false,
    walk:0, ext:0, ph:Math.random()*6, spawnT:16,
    limbs:[]
  };
  d.limbs.forEach(function(L){
    e.limbs.push({ id:L.id, ox:L.ox, oy:L.oy, r:L.r, hp:L.hp, maxHp:L.hp, core:!!L.core,
      broken:false, x:e.x+L.ox, y:e.y+L.oy, ext:0 });
  });
  G.enemies.push(e);
  addFx({t:'ring', x:e.x, y:e.y-e.h*0.5, r:6, vr:6, life:18, max:18, color:'rgba(255,90,120,1)', w:4});
  burst(e.x, e.y-e.h*0.5, 10, '#ff7a95');
  return e;
}

function armsOf(e, ids){
  return e.limbs.filter(function(L){ return !L.broken && ids.indexOf(L.id) >= 0; }).length;
}

function updateEnemy(e){
  e.t++;
  if(e.hitFlash > 0) e.hitFlash--;
  if(e.atkCd > 0) e.atkCd--;
  if(e.spawnT > 0) e.spawnT--;
  e.walk = 0; e.ext = 0;

  var dx = P.x - e.x, adx = Math.abs(dx);

  if(e.stun > 0){
    e.stun--;
    if(e.stun === 0 && e.st === 'wind'){ e.st = 'idle'; e.t = 0; e.atkCd = 40; }
  } else if(!P.dead){
    if(e.type === 'soldier') aiSoldier(e, dx, adx);
    else if(e.type === 'gunner') aiGunner(e, dx, adx);
    else aiBrute(e, dx, adx);
  }

  // 物理
  e.vy += 0.7;
  e.x += e.walk * e.face + e.vx;
  e.y += e.vy;
  if(e.y >= GROUND){
    if(e.vy > 8) puff(e.x, GROUND, 6, '#8c7fb0');
    e.y = GROUND; e.vy = 0;
  }
  e.vx *= 0.87;
  e.x = clamp(e.x, 40, ARENA_W-40);

  // 肢体位置
  var i, L;
  for(i=0;i<e.limbs.length;i++){
    L = e.limbs[i];
    if(L.broken) continue;
    L.x = e.x + (L.ox + L.ext) * e.face;
    L.y = e.y + L.oy;
  }
  e.ph += e.walk * 0.11 + 0.012;

  if(e.hp <= 0 && !e.dead) killEnemy(e);
}

function aiSoldier(e, dx, adx){
  e.face = dx > 0 ? 1 : -1;
  var armN = armsOf(e, ['armR','armL']);
  var range = armN > 0 ? 96 : 62;

  if(e.st === 'idle' || e.st === 'chase'){
    if(adx > range){ e.walk = e.speed; e.st = 'chase'; }
    else if(e.atkCd <= 0){ e.st = 'wind'; e.t = 0; }
    else { e.walk = 0; }
    if(adx > 420 && e.atkCd <= 0) e.walk = e.speed * 1.4;
  } else if(e.st === 'wind'){
    e.walk = adx > range + 24 ? e.speed*0.5 : 0;
    var wt = armN > 0 ? 34 : 26;
    var pr = Math.min(1, e.t/wt);
    e.limbs.forEach(function(L){ if(!L.broken && !L.core) L.ext = -12*pr; });
    if(e.t >= wt){
      e.st = 'swing'; e.t = 0;
      var reach = armN > 0 ? 96 : 54;
      var yTop = e.y - (armN > 0 ? 104 : 70);
      addHazard({ x: e.x + (e.face>0? 10 : -10-reach), y:yTop, w:reach, h: (armN>0?84:56),
        dmg: armN > 0 ? 12 : 6, owner:e, life:9 });
      noiseHit(0.14, 1600, 300, 0.14, 1.2);
    }
  } else if(e.st === 'swing'){
    e.walk = 0;
    e.limbs.forEach(function(L){ if(!L.broken && !L.core) L.ext = 26; });
    if(e.t >= 11){ e.st = 'recover'; e.t = 0; }
  } else if(e.st === 'recover'){
    if(e.t >= (armN > 0 ? 30 : 44)){ e.st = 'idle'; e.t = 0; e.atkCd = 28 + Math.random()*40; }
  }
}

function aiGunner(e, dx, adx){
  e.face = dx > 0 ? 1 : -1;
  var hasCannon = armsOf(e, ['cannon']) > 0;

  if(e.st === 'idle' || e.st === 'chase'){
    if(hasCannon){
      if(adx < 240) e.walk = -e.speed*1.05;
      else if(adx > 440) e.walk = e.speed*0.85;
      if(e.atkCd <= 0 && adx < 620 && e.spawnT <= 0){ e.st = 'aim'; e.t = 0; }
    } else {
      if(adx > 76) e.walk = e.speed*1.1;
      else if(e.atkCd <= 0){ e.st = 'wind'; e.t = 0; }
    }
  } else if(e.st === 'aim'){
    e.walk = adx < 190 ? -e.speed*0.5 : 0;
    var L = e.limbs.find(function(l){ return l.id === 'cannon' && !l.broken; });
    if(L) L.ext = 10 * Math.min(1, e.t/40);
    if(e.t >= 42){
      e.st = 'fire'; e.t = 0;
      var mx = e.x + e.face*26, my = e.y - 64;
      var ang = Math.atan2((P.y-58) - my, P.x - mx);
      G.proj.push({ x:mx, y:my, vx:Math.cos(ang)*6.2, vy:Math.sin(ang)*6.2, r:9, dmg:10, life:150, dead:false, owner:e });
      SFX.shoot();
      addFx({t:'flash', x:mx, y:my, r:26, life:7, max:7, color:'rgba(255,120,150,'});
      burst(mx, my, 7, '#ff8fa8');
      e.vx -= e.face*2.4;
    }
  } else if(e.st === 'fire'){
    if(e.t >= 20){ e.st = 'recover'; e.t = 0; }
  } else if(e.st === 'wind'){
    e.walk = 0;
    if(e.t >= 24){
      e.st = 'swing'; e.t = 0;
      addHazard({ x: e.x + (e.face>0? 10 : -64), y: e.y-70, w:54, h:52, dmg:6, owner:e, life:8 });
    }
  } else if(e.st === 'swing'){
    if(e.t >= 9){ e.st = 'recover'; e.t = 0; }
  } else if(e.st === 'recover'){
    if(e.t >= 40){ e.st = 'idle'; e.t = 0; e.atkCd = 34 + Math.random()*46; }
  }
}

function aiBrute(e, dx, adx){
  e.face = dx > 0 ? 1 : -1;
  if(e.st === 'idle' || e.st === 'chase'){
    if(adx > 140){ e.walk = e.speed; e.st = 'chase'; }
    else if(e.atkCd <= 0){ e.st = 'wind'; e.t = 0; }
  } else if(e.st === 'wind'){
    e.walk = adx > 170 ? e.speed*0.6 : 0;
    e.limbs.forEach(function(L){ if(!L.broken && !L.core) L.ext = -18*Math.min(1,e.t/48); });
    if(e.t >= 48){
      e.st = 'swing'; e.t = 0;
      addHazard({ x:e.x-150, y:e.y-96, w:300, h:96, dmg:16, owner:e, life:12 });
      G.shake = Math.max(G.shake, 8);
      addFx({t:'ring', x:e.x, y:GROUND-6, r:10, vr:14, life:24, max:24, color:'rgba(255,170,90,1)', w:7});
      puff(e.x-60, GROUND, 10, '#b09a80'); puff(e.x+60, GROUND, 10, '#b09a80');
      noiseHit(0.3, 500, 90, 0.3, 0.7);
    }
  } else if(e.st === 'swing'){
    if(e.t >= 14){ e.st = 'recover'; e.t = 0; }
  } else if(e.st === 'recover'){
    if(e.t >= 46){ e.st = 'idle'; e.t = 0; e.atkCd = 50 + Math.random()*40; }
  }
}

/* ============================== 弹幕 / 危险区 ============================== */
function addHazard(o){
  o.life = o.life || 10;
  o.t = 0; o.hitP = false;
  G.haz.push(o);
}

function updateHazards(){
  var i, h;
  var pbox = { x:P.x-15, y:P.y-84, w:30, h:84 };
  for(i=G.haz.length-1;i>=0;i--){
    h = G.haz[i];
    h.life--; h.t++;
    if(!h.hitP && !P.dead){
      if(h.x < pbox.x+pbox.w && h.x+h.w > pbox.x && h.y < pbox.y+pbox.h && h.y+h.h > pbox.y){
        var dir = sign(P.x - (h.x + h.w/2)) || 1;
        if(hurtPlayer(h.dmg, dir, h.owner)){ h.hitP = true; }
      }
    }
    if(h.life <= 0) G.haz.splice(i,1);
  }
}

function updateProjectiles(){
  var i, pr;
  for(i=G.proj.length-1;i>=0;i--){
    pr = G.proj[i];
    pr.life--;
    pr.x += pr.vx; pr.y += pr.vy;
    if(pr.life <= 0 || pr.x < 0 || pr.x > ARENA_W) { G.proj.splice(i,1); continue; }
    if(pr.dead){ G.proj.splice(i,1); continue; }
    // 见切反弹
    if(P.state === 'parry' && P.t >= 3 && P.t <= 16 && Math.hypot(pr.x-P.x, pr.y-(P.y-58)) < 90){
      doParry(pr.owner);
      G.proj.splice(i,1);
      continue;
    }
    if(pr.y > GROUND){ G.proj.splice(i,1); puff(pr.x, GROUND, 6, '#ff8fa8'); continue; }
    if(!P.dead && P.invuln <= 0 && P.state !== 'dash'){
      if(pr.x > P.x-16 && pr.x < P.x+16 && pr.y > P.y-86 && pr.y < P.y){
        if(hurtPlayer(pr.dmg, sign(pr.vx)||1, pr.owner)){ G.proj.splice(i,1); continue; }
      }
    }
    if(Math.random() < 0.5) addFx({t:'spark', x:pr.x, y:pr.y, vx:rnd(-1,1), vy:rnd(-1,1),
      life:12, max:12, color:'#ff8fa8', size:2});
  }
}

/* ============================== 波次 ============================== */
function spawnWave(){
  G.wave++;
  G.waveMsg = 'WAVE ' + G.wave;
  G.waveMsgT = 110;
  SFX.wave();
  var w = G.wave;
  var comp = [];
  comp.push('soldier');
  if(w >= 2) comp.push(Math.random() < 0.55 ? 'soldier' : 'gunner');
  if(w >= 3) comp.push('gunner');
  if(w >= 4) comp.push('soldier');
  if(w >= 5 && w % 2 === 1) comp.push('brute');
  if(w >= 7) comp.push('gunner');
  if(comp.length > 5) comp.length = 5;

  var side = 1;
  for(var i=0;i<comp.length;i++){
    var dist = 380 + Math.random()*260;
    side = (i % 2 === 0) ? 1 : -1;
    var x = P.x + side*dist;
    if(x < 120 || x > ARENA_W-120){ x = P.x - side*dist; }
    x = clamp(x, 100, ARENA_W-100);
    spawnEnemy(comp[i], x);
  }
}

function updateCamera(){
  var tx = P.x + P.face*64;
  var ty = 336 + (P.y - GROUND)*0.32;
  G.camX = lerp(G.camX, tx, 0.11);
  G.camY = lerp(G.camY, ty, 0.09);
  var half = W/2;
  var lo = half, hi = Math.max(half, ARENA_W - half);
  G.camX = clamp(G.camX, lo, hi);
  G.camY = clamp(G.camY, 250, 470);
}

/* ============================== 主循环 ============================== */
function step(){
  FT++;
  G.shx = (Math.random()-0.5)*G.shake;
  G.shy = (Math.random()-0.5)*G.shake;
  G.shake *= 0.87;
  if(G.shake < 0.25) G.shake = 0;
  if(G.hurtFlash > 0) G.hurtFlash--;
  if(G.waveMsgT > 0) G.waveMsgT--;

  var hs = G.hitstop > 0;
  if(hs) G.hitstop--;

  var k = hs ? 0.32 : 1;

  if(G.state === 'play'){
    if(!hs){
      updatePlayer();
      for(var i=G.enemies.length-1;i>=0;i--){
        if(G.enemies[i]) updateEnemy(G.enemies[i]);
      }
      updateHazards();
      updateProjectiles();
      G.hpGhost = lerp(G.hpGhost, P.hp, 0.12);
      // 波次推进
      if(G.enemies.length === 0){
        if(G.waveT <= 0) G.waveT = 76;
        G.waveT--;
        if(G.waveT <= 0 && G.wave === 0) spawnWave();
        else if(G.waveT <= 0 && G.enemies.length === 0) spawnWave();
      } else {
        G.waveT = 0;
      }
    }
  } else {
    G.hpGhost = lerp(G.hpGhost, P.hp, 0.12);
  }

  updateFx(k);
  updateCamera();

  // 环境余烬
  if(G.embers.length < 46 && Math.random() < 0.55){
    G.embers.push({ x:G.camX + rnd(-W*0.6, W*0.6), y:G.camY + rnd(-200, 320),
      vx:rnd(-0.35,0.35), vy:rnd(-0.9,-0.2), life:rnd(120,320), max:320, s:rnd(0.8,2.2) });
  }
  for(var m=G.embers.length-1;m>=0;m--){
    var em = G.embers[m];
    em.life--;
    em.x += em.vx; em.y += em.vy;
    if(em.life <= 0) G.embers.splice(m,1);
  }
  if(buf.light > 0) buf.light--;
  if(buf.heavy > 0) buf.heavy--;
  if(buf.dash > 0) buf.dash--;
  if(buf.parry > 0) buf.parry--;
  if(buf.jump > 0) buf.jump--;
}

var lastT = 0, acc = 0;
function loop(now){
  requestAnimationFrame(loop);
  if(!lastT) lastT = now;
  var dt = (now - lastT)/1000;
  lastT = now;
  if(dt > 0.12) dt = 0.12;
  acc += dt;
  var guard = 0;
  while(acc >= 1/60 && guard < 5){ acc -= 1/60; step(); guard++; }
  if(guard >= 5) acc = 0;
  render();
}

/* ---------- 玩家姿态 ---------- */
function getPose(){
  var p = P, s = p.state, t = p.t;
  var hipY = -46, footA = [10,0], footB = [-10,0], hand = [11,-64], sAng = 1.0, lean = 0, hipX = 0;
  var headTilt = 0;

  if(s === 'idle'){
    var bob = Math.sin(G.time*0.05)*1.4;
    hipY = -46 + bob;
    hand = [12, -64 + bob];
    sAng = 1.05;
  } else if(s === 'run'){
    var ph = G.time*0.36;
    hipY = -47 + Math.abs(Math.sin(ph))*1.6;
    footA = [Math.cos(ph)*16, -Math.max(0, Math.sin(ph))*15];
    footB = [Math.cos(ph+Math.PI)*16, -Math.max(0, Math.sin(ph+Math.PI))*15];
    hand = [9, -63];
    sAng = 1.25;
    lean = -0.13;
  } else if(s === 'jump'){
    footA = [11,-20]; footB = [-7,-27];
    hand = [8,-70]; sAng = 0.7; lean = -0.08;
  } else if(s === 'fall'){
    footA = [13,-10]; footB = [-9,-16];
    hand = [13,-66]; sAng = 1.1; lean = 0.06;
  } else if(s === 'dash'){
    hipY = -42; hipX = 4; lean = 0.34;
    footA = [-6,-16]; footB = [-20,-8];
    hand = [-4,-58]; sAng = 0.35;
  } else if(s === 'parry'){
    var q = Math.min(1, P.t/6);
    hand = [lerp(12,20,q), lerp(-64,-76,q)];
    sAng = lerp(1.05, -1.45, q);
    footA = [12,0]; footB = [-12,0];
    lean = -0.12*q;
  } else if(s === 'hurt'){
    hipY = -44; lean = -0.3;
    footA = [8,0]; footB = [-14,0];
    hand = [-2,-58]; sAng = 2.2;
  } else if(s === 'charge'){
    var c = P.charge/36;
    hipY = -44; lean = -0.16 + c*0.1;
    footA = [14,0]; footB = [-14,0];
    hand = [lerp(-2,-12,c), lerp(-74,-96,c)];
    sAng = lerp(-2.4, -3.05, c);
    headTilt = -0.1;
  } else if(s === 'dead'){
    hipY = -30; lean = 0.9;
    footA = [12,-4]; footB = [-12,-2];
    hand = [10,-30]; sAng = 1.7;
  } else if(p.cur){
    var a = p.cur, k0 = a.k[0], k1 = a.k[1], k2 = a.k[2];
    var fr;
    if(p.phase === 'startup'){
      fr = ease(clamp(p.pt/a.s, 0, 1));
      hand = [lerp(k0[0],k1[0],fr), lerp(k0[1],k1[1],fr)];
      sAng = lerp(k0[2],k1[2],fr);
      lean = -0.10*fr;
      footA = [10,0]; footB = [-11,0]; hipY = -46;
    } else if(p.phase === 'active'){
      fr = ease(clamp(p.pt/a.a, 0, 1));
      hand = [lerp(k1[0],k2[0],fr), lerp(k1[1],k2[1],fr)];
      sAng = lerp(k1[2],k2[2],fr);
      lean = 0.16*fr;
      footA = [13,0]; footB = [-13,0]; hipY = -45;
    } else {
      fr = clamp(p.pt/a.r, 0, 1);
      hand = [lerp(k2[0],11,fr), lerp(k2[1],-64,fr)];
      sAng = lerp(k2[2],1.05,fr);
      lean = lerp(0.16, 0, fr);
      footA = [10,0]; footB = [-10,0]; hipY = -46;
    }
    if(!P.onGround){ footA = [10,-18]; footB = [-8,-24]; }
  }
  return { hip:[hipX,hipY], footA:footA, footB:footB, hand:hand, sAng:sAng, lean:lean, headTilt:headTilt };
}
function ease(t){ return t<0.5 ? 2*t*t : -1+(4-2*t)*t; }

function ik(ax,ay,bx,by,l1,l2,bend){
  var dx = bx-ax, dy = by-ay;
  var d = Math.hypot(dx,dy);
  d = Math.min(d, l1+l2-0.5) || 0.01;
  var a = Math.atan2(dy,dx);
  var cosa = (l1*l1 + d*d - l2*l2)/(2*l1*d);
  var ang = Math.acos(clamp(cosa,-1,1));
  return [ax + Math.cos(a + bend*ang)*l1, ay + Math.sin(a + bend*ang)*l1];
}

function limbLine(a, b, w, color){
  ctx.strokeStyle = color;
  ctx.lineWidth = w;
  ctx.lineCap = 'round';
  ctx.beginPath();

  ctx.stroke();
}

  ctx.beginPath();
  ctx.moveTo(hip[0]-11, hip[1]);
  ctx.lineTo(hip[0]+11, hip[1]);

  ctx.lineTo(shoulder[0]-10, shoulder[1]-2);
  ctx.closePath();

  ctx.moveTo(hip[0]-12, hip[1]-4);
  ctx.lineTo(hip[0]+12, hip[1]-4);
  ctx.lineTo(hip[0]+16, hip[1]+18);

  ctx.fill();

  if(!ghost){
    ctx.fillStyle = '#7fe4ff';
    ctx.beginPath(); ctx.arc(hx+4, hy+1, 1.9, 0, 6.283); ctx.fill();
  }

  // 前臂
  var elbowA = ik(shoulder[0], shoulder[1], hand[0], hand[1], 18, 18, -1);
  limbLine(shoulder, elbowA, 7.5, ghost?'#6f8bd0':'#2a3352');
  limbLine(elbowA, hand, 6.5, ghost?'#8aa6e6':'#3d4a72');

  // 刀
  var sLen = 58;
  var tx = hand[0] + Math.cos(pose.sAng)*sLen;
  var ty = hand[1] + Math.sin(pose.sAng)*sLen;
  ctx.save();
  ctx.lineCap = 'round';
  if(!ghost){
    ctx.globalCompositeOperation = 'lighter';
    ctx.strokeStyle = 'rgba(130,220,255,0.30)';
    ctx.lineWidth = 9;
    ctx.beginPath(); ctx.moveTo(hand[0], hand[1]); ctx.lineTo(tx, ty); ctx.stroke();
    ctx.globalCompositeOperation = 'source-over';
  }
  ctx.strokeStyle = ghost ? 'rgba(170,200,255,0.5)' : '#dff3ff';
  ctx.lineWidth = 3.6;
  ctx.beginPath(); ctx.moveTo(hand[0], hand[1]); ctx.lineTo(tx, ty); ctx.stroke();
  if(!ghost){
    ctx.strokeStyle = '#8e6a3c';
    ctx.lineWidth = 5;
    ctx.beginPath(); ctx.moveTo(hand[0], hand[1]);
    ctx.lineTo(hand[0]-Math.cos(pose.sAng)*8, hand[1]-Math.sin(pose.sAng)*8); ctx.stroke();
  }

  ctx.restore();

function drawEnemy(e){
  var full = e.type === 'brute';

    ctx.beginPath(); ctx.arc(0, headY, full?17:13, 0, 6.283); ctx.fill();
    ctx.strokeStyle = hurt ? '#fff' : '#7c5836';

      ctx.beginPath(); ctx.arc(5, headY-2, 3.4, 0, 6.283); ctx.fill();
      ctx.fillStyle = '#ff4d6d';

      ctx.beginPath(); ctx.arc(-4, headY-2, 1.7*gl, 0, 6.283); ctx.fill();
      ctx.beginPath(); ctx.arc(5, headY-2, 1.7*gl, 0, 6.283); ctx.fill();
      // 提线
      ctx.strokeStyle = 'rgba(255,120,150,0.22)';
      ctx.lineWidth = 1;
      ctx.beginPath(); ctx.moveTo(0, headY-13); ctx.lineTo(6, headY-70); ctx.stroke();
    }
  }

  // 攻击预兆
  if(e.st === 'wind'){
    var kk = Math.min(1, e.t/30);
    ctx.save();
    ctx.globalAlpha = 0.18 + 0.42*kk;
    ctx.fillStyle = '#ff3b5c';
    ctx.beginPath();
    if(e.type === 'brute'){
      ctx.moveTo(-150, -96); ctx.lineTo(150, -96); ctx.lineTo(150, 0); ctx.lineTo(-150, 0);

    }
    ctx.closePath(); ctx.fill();
    ctx.restore();
  } else if(e.st === 'aim'){
    var kk2 = Math.min(1, e.t/42);
    ctx.save();
    ctx.globalAlpha = 0.25 + 0.45*kk2;
    ctx.strokeStyle = '#ff3b5c';
    ctx.lineWidth = 1.5 + kk2*1.5;

    ctx.setLineDash([]);
    ctx.restore();
  }

  ctx.restore();

  // 血条
  if(!e.dead && e.hp < e.maxHp){
    var bw = Math.max(34, e.w*1.5), bh = 4;
    var bx = e.x - bw/2, by = e.y - e.h - 16;
    ctx.fillStyle = 'rgba(0,0,0,0.55)';
    ctx.fillRect(bx, by, bw, bh);
    ctx.fillStyle = '#ff5f7e';
    ctx.fillRect(bx, by, bw*clamp(e.hp/e.maxHp,0,1), bh);
  }
  // 连击提示用的命中圈
}

    g.addColorStop(1, 'rgba(255,60,110,0)');
    ctx.fillStyle = g;
    ctx.beginPath(); ctx.arc(p.x, p.y, p.r*3, 0, 6.283); ctx.fill();
    ctx.restore();
  }
}

function drawArcs(){
  for(var i=0;i<G.arcs.length;i++){
    var f = G.arcs[i];
    var k = f.life/f.max;
    var a0 = Math.min(f.a0, f.a1), a1 = Math.max(f.a0, f.a1);
    ctx.save();
    ctx.globalCompositeOperation = 'lighter';
    ctx.translate(f.x, f.y);
    ctx.scale(f.face, 1);
    ctx.globalAlpha = k*0.85;
    ctx.strokeStyle = f.color;
    ctx.lineWidth = f.w*(0.45 + k*0.75);
    ctx.lineCap = 'round';
    ctx.beginPath(); ctx.arc(0, 0, f.r*(1 + (1-k)*0.14), a0, a1); ctx.stroke();
    ctx.globalAlpha = k;
    ctx.strokeStyle = '#ffffff';
    ctx.lineWidth = 2.6;
    ctx.beginPath(); ctx.arc(0, 0, f.r*(1 + (1-k)*0.14) - 3, a0, a1); ctx.stroke();
    ctx.restore();
  }
}

function drawFx(){
  for(var i=0;i<G.fx.length;i++){
    var f = G.fx[i];
    var k = f.life/f.max;
    if(f.t === 'spark'){
      ctx.save();
      ctx.globalCompositeOperation = 'lighter';
      ctx.globalAlpha = Math.min(1, k*1.4);
      ctx.strokeStyle = f.color;
      ctx.lineWidth = f.size;
      ctx.lineCap = 'round';
      ctx.beginPath();
      ctx.moveTo(f.x, f.y);
      ctx.lineTo(f.x - f.vx*1.6, f.y - f.vy*1.6);
      ctx.stroke();
      ctx.restore();
    } else if(f.t === 'shard'){
      ctx.save();
      ctx.globalAlpha = Math.min(1, k*1.6);
      ctx.translate(f.x, f.y);
      ctx.rotate(f.rot);
      ctx.fillStyle = f.color;
      ctx.fillRect(-f.size*0.5, -f.size*0.22, f.size, f.size*0.44);
      ctx.restore();
    } else if(f.t === 'ring'){
      ctx.save();
      ctx.globalCompositeOperation = 'lighter';
      ctx.globalAlpha = k*0.9;
      ctx.strokeStyle = f.color;
      ctx.lineWidth = f.w*k;
      ctx.beginPath(); ctx.arc(f.x, f.y, f.r, 0, 6.283); ctx.stroke();
      ctx.restore();
    } else if(f.t === 'flash'){
      ctx.save();
      ctx.globalCompositeOperation = 'lighter';
      ctx.globalAlpha = k*0.9;
      var g = ctx.createRadialGradient(f.x, f.y, 1, f.x, f.y, f.r);
      g.addColorStop(0, f.color + (k*0.9) + ')');
      g.addColorStop(1, f.color + '0)');
      ctx.fillStyle = g;
      ctx.beginPath(); ctx.arc(f.x, f.y, f.r, 0, 6.283); ctx.fill();
      ctx.restore();
    }
  }
}
