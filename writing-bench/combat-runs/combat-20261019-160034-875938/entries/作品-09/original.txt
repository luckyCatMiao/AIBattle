/* ===================== 波次 ===================== */
function startWave(){
  G.wave++;
  const w = G.wave;
  const q = [];
  if(w===1){ q.push('puppet','puppet'); }
  else if(w===2){ q.push('puppet','dasher','puppet'); }
  else if(w===3){ q.push('dasher','puppet','dasher','puppet'); }
  else {
    const n = Math.min(6, 3 + Math.floor(w/2));
    for(let i=0;i<n;i++) q.push(Math.random()<0.45?'dasher':'puppet');
  }
  G.waveQueue = q;
  G.spawnTimer = 0.12;
  G.betweenWaves = 0;
  addText(player.x, player.y-120, '第 '+w+' 波', '#9fd4ff', 26);
  if(w>1){ player.hp = Math.min(player.maxHp, player.hp+15); }
}

    parts: k.parts.map(mkPart),
    state:'spawn', t:0, atkCd: rand(0.5,1.1),
    flash:0, hitStun:0, staggerT:0.45,
    dead:false, deadT:0, hasHit:false, walkPh:rand(0,6),
    armA:1.1, eye:0
  };
  G.enemies.push(e);
  addRing(x, GY-34, 8, 64, 0.42, '#7fd0ff');
  addSpark(x, GY-34, 12, '#8fd8ff', 260);
  sfx('spawn');
}
function hasPart(e,id){
  for(let i=0;i<e.parts.length;i++) if(e.parts[i].id===id) return !e.parts[i].broken;
  return false;
}

/* ===================== 攻防 ===================== */
function playerRect(){
  return {x:player.x-14, y:player.y-76, w:28, h:76};
}

function beginAttack(name){
  const p = player;
  let a;
  if(name==='heavy'){
    const c = clamp(p.chargeT/1.0,0,1);
    a = {
      dur:0.44, hs:0.11, he:0.22,
      dmg: 20+28*c, limb: 999, kb: 320+380*c,
      lift: c>0.7? -300 : -60,
      stop: 0.085+0.065*c, shake: 6+7*c,
      reach: 84, hh: 74, hy:-50,
      a0:-2.35, a1:1.65, r:82,
      next:null, dashF: 220+180*c, col:'#ffe9b0',
      heavy:true, charge:c
    };
  } else {
    a = Object.assign({}, ATK[name]);
  }
  a.name = name;
  p.state='attack'; p.atk=a; p.t=0; p.hitSet=new Set(); p.buffer=null;
  if(a.dashF) p.vx += p.facing*a.dashF;
  addSlash(p.x+p.facing*10, p.y-46, a.a0, a.a1, a.r, a.col, name==='heavy'?0.34:0.22, name==='heavy'?26:15, p.facing);
  if(name==='heavy'){
    addRing(p.x, p.y-46, 10, 90, 0.3, '#ffd98a');
    sfx('swing2',1);
  } else {
    sfx('swing',0.9);
  }
}

function makeHeavyAttack(){ return 'heavy'; }

function doAttackHit(a){
  const p = player;
  const w = a.reach, h = a.hh;
  const cx = p.x + p.facing*(w*0.5+14);
  const hb = {x:cx-w/2, y:p.y+a.hy-h/2, w:w, h:h};
  for(let i=0;i<G.enemies.length;i++){
    const e = G.enemies[i];
    if(e.dead || e.hp<=0) continue;
    if(p.hitSet.has(e.uid)) continue;
    if(hitEnemy(e, hb, a)){
      p.hitSet.add(e.uid);
    }
  }
}

function hitEnemy(e, hb, a){
  const f = e.facing;
  // 先检测肢体
  let target = null;
  for(let i=0;i<e.parts.length;i++){
    const pt = e.parts[i];
    if(pt.broken) continue;
    const px = e.x + pt.ox*f, py = e.y + pt.oy;
    if(circleRect(px,py,pt.r+3,hb)){ target = pt; break; }
  }
  const mul = 1 + Math.min(player.combo,20)*0.02;
  if(target){
    const dmg = a.limb*mul;
    target.hp -= dmg;
    e.flash = 0.13;
    e.hitStun = Math.max(e.hitStun, 0.14);
    const hx = e.x + target.ox*f, hy = e.y + target.oy;
    addSpark(hx, hy, 10, '#fff2c0', 380, Math.atan2(hy-(player.y-50), hx-player.x));
    addText(hx, hy-16, '-'+Math.round(a.dmg*mul*0.5), '#ffe6a0', 13);
    hitStop(a.stop*0.8); addShake(a.shake*0.7);
    e.vx += player.facing*a.kb*0.3;
    G.hitCount++;
    player.combo++; player.comboT = 2.6;
    sfx('hit', 0.75);
    if(target.hp<=0){ breakPart(e, target); }
    return true;
  }
  const core = {x:e.x-19, y:e.y-e.h, w:38, h:e.h-4};
  if(rectRect(core, hb)){
    const dmg = a.dmg*mul;
    e.hp -= dmg;
    e.flash = 0.14;
    e.hitStun = Math.max(e.hitStun, 0.2);
    e.vx += player.facing*a.kb;
    if(a.lift){ e.vy = a.lift*0.9; e.onGround=false; }
    const hx = player.x + player.facing*40, hy = player.y-52;
    addSpark(hx, hy, a.heavy?18:11, a.heavy?'#ffe0a0':'#ffffff', a.heavy?620:400, player.facing>0?0:Math.PI);
    addRing(hx, hy, 6, a.heavy?90:52, 0.28, a.heavy?'#ffd98a':'#bfe9ff');
    addText(hx, hy-24, Math.round(dmg), a.heavy?'#ffd479':'#eaf6ff', a.heavy?20:14);
    hitStop(a.stop); addShake(a.shake);
    G.hitCount++;
    player.combo++; player.comboT = 2.6;
    sfx(a.heavy?'heavy':'hit', a.heavy?1:0.85);
    if(e.hp<=0) killEnemy(e);
    return true;
  }
  return false;
}

function breakPart(e, pt){
  if(pt.broken) return;
  pt.broken = true;
  G.limbBreakCount++;
  const f = e.facing;
  const px = e.x + pt.ox*f, py = e.y + pt.oy;
  const col = e.kind==='puppet' ? '#7a5334' : '#5b3c6b';
  const col2 = e.kind==='puppet' ? '#d9c9a8' : '#8f6aa8';
  for(let i=0;i<9;i++) addDebris(px, py, i%3===0?col2:col, 420, pt.r*0.9);
  addSpark(px, py, 22, '#ff8a6a', 520, 0);
  addSpark(px, py, 12, '#fff0c0', 340, 0);
  addRing(px, py, 8, 78, 0.35, '#ff9a6a');
  addText(px, py-26, '断 '+pt.label+'!', '#ff9d6a', 18);
  hitStop(0.115); addShake(7.5);
  sfx('break');
  e.staggerT = Math.max(e.staggerT, 0.62);
  e.hitStun = Math.max(e.hitStun, 0.3);
}

function killEnemy(e){
  if(e.dead) return;
  e.dead = true; e.deadT = 0;
  e.vy = -300; e.vx = player.facing*260; e.onGround = false;
  const col = e.kind==='puppet' ? '#6b4a30' : '#5b3c6b';
  for(let i=0;i<16;i++) addDebris(e.x+rand(-10,10), e.y-e.h*0.5+rand(-16,16), col, 520, rand(3,7));
  addSpark(e.x, e.y-e.h*0.5, 26, '#ffd0a0', 560, 0);
  addRing(e.x, e.y-e.h*0.5, 10, 120, 0.45, '#ffc98a');
  hitStop(0.085); addShake(6.5);
  sfx('die');
  G.kills++;
  G.hitCount++;
  player.combo++; player.comboT = 2.6;
  addText(e.x, e.y-e.h-14, '粉碎', '#ffcf8a', 17);
}

function damagePlayer(dmg, fromX){
  const p = player;
  if(p.state==='dead') return;
  p.hp -= dmg;
  p.hurtFlash = 0.22;
  p.invuln = 0.75;
  G.hurtVig = 1;
  const dir = p.x < fromX ? -1 : 1;
  p.vx = dir*280; p.vy = -220; p.onGround = false;
  p.state = 'hurt'; p.t = 0; p.atk = null;
  p.combo = 0; p.comboT = 0;
  hitStop(0.06); addShake(6);
  addSpark(p.x, p.y-50, 14, '#ff5a5a', 420, dir>0?0:Math.PI);
  sfx('hurt');
  if(p.hp<=0){
    p.hp = 0; p.state='dead'; p.deadT=0; p.dead=true;
    G.over = true;
    sfx('die');
    addRing(p.x, p.y-40, 10, 150, 0.6, '#ff5555');
    setTimeout(showDeath, 850);
  }
}

function doParry(e){
  const p = player;
  hitStop(0.24);
  G.slowmo = 0.5;
  addShake(8);
  sfx('parry');
  addSlash(p.x+p.facing*12, p.y-48, -2.2, 1.7, 92, '#cfeeff', 0.42, 22, p.facing);
  addRing(p.x, p.y-46, 8, 130, 0.5, '#9fe0ff');
  addSpark(p.x+p.facing*34, p.y-50, 26, '#ffffff', 560, p.facing>0?0:Math.PI);
  addText(p.x, p.y-104, '弹 反!', '#9fe0ff', 24);
  if(e){
    e.staggerT = 1.25; e.flash = 0.22; e.hitStun = 0.5;
    e.vx += p.facing*220;
    const intact = e.parts.filter(function(q){ return !q.broken; });
    if(intact.length){ breakPart(e, intact[0]); }
    else { e.hp -= 26; if(e.hp<=0) killEnemy(e); }
  }
  G.hitCount++;
  p.combo++; p.comboT = 2.6;
  p.state = 'idle'; p.t = 0;
}

/* ===================== 玩家更新 ===================== */
function updatePlayer(dt){
  const p = player;
  p.t += dt;
  p.invuln = Math.max(0,p.invuln-dt);
  p.iframe = Math.max(0,p.iframe-dt);
  p.dashCd = Math.max(0,p.dashCd-dt);
  p.parryCd = Math.max(0,p.parryCd-dt);
  p.hurtFlash = Math.max(0,p.hurtFlash-dt);
  if(p.bufferT>0){ p.bufferT -= dt; if(p.bufferT<=0) p.buffer=null; }
  if(p.comboT>0){ p.comboT -= dt; if(p.comboT<=0) p.combo=0; }
  if(p.state!=='dead') p.hpLag = lerp(p.hpLag, p.hp, 1-Math.pow(0.0015,dt));

  if(p.state==='dead'){
    p.deadT += dt;
    p.vy += GRAV*dt; p.y += p.vy*dt; p.x += p.vx*dt;
    p.vx = approach(p.vx,0,1200*dt);
    if(p.y>GY){ p.y=GY; p.vy=0; }
    return;
  }

  const moveDir = (keys.right?1:0) - (keys.left?1:0);

  /* ---- 状态处理 ---- */
  if(p.state==='idle' || p.state==='run' || p.state==='air'){
    if(moveDir!==0){ p.facing = moveDir; p.vx = approach(p.vx, moveDir*SPEED, ACC*dt); }
    else p.vx = approach(p.vx, 0, FRIC*dt);
    if(p.onGround) p.state = Math.abs(p.vx)>25 ? 'run' : 'idle';
    else p.state = 'air';
  } else if(p.state==='attack'){
    const a = p.atk;
    p.vx = approach(p.vx, 0, 1500*dt);
    if(p.t>=a.hs && p.t<=a.he) doAttackHit(a);
    if(p.buffer==='atk' && p.t >= a.dur*0.52){
      if(a.next){ beginAttack(a.next); p.buffer=null; }
      else if(p.onGround){ beginAttack('l1'); p.buffer=null; }
      else if(p.airAtk < 2){ beginAttack('air1'); p.buffer=null; }
    }
    if(p.t>=a.dur){
      p.state = p.onGround?'idle':'air';
      p.atk = null; p.buffer = null;
    }
  } else if(p.state==='charge'){
    p.chargeT = Math.min(1.0, p.chargeT + dt);
    p.vx = approach(p.vx, moveDir*SPEED*0.35, ACC*dt);
    if(moveDir!==0) p.facing = moveDir;
    if(!keys.heavy){ beginAttack('heavy'); }
  } else if(p.state==='dash'){
    p.iframe = Math.max(p.iframe, 0.02);
    if(p.t >= p.dashDur){
      p.state = p.onGround?'idle':'air';
    }
  } else if(p.state==='parry'){
    if(p.t >= 0.42){ p.state = p.onGround?'idle':'air'; }
  } else if(p.state==='hurt'){
    p.vx = approach(p.vx, 0, 900*dt);
    if(p.t >= 0.3){ p.state = p.onGround?'idle':'air'; }
  }

  /* ---- 输入 ---- */
  if(pressed.jump){
    if(p.state!=='attack' && p.state!=='dash' && p.state!=='charge'){
      if(p.onGround){
        p.vy = JUMPV; p.onGround=false; p.jumps=1; p.state='air';
        p.airDash = 1; p.airAtk = 0;
        addDust(p.x, GY, 6); sfx('jump');
      } else if(p.jumps<2){
        p.vy = JUMPV*0.88; p.jumps=2; p.state='air';
        p.airDash = 1; p.airAtk = 0;
        addRing(p.x, p.y-30, 6, 44, 0.3, '#9fd8ff');
        addSpark(p.x, p.y-20, 10, '#9fd8ff', 220, Math.PI*0.5);
        sfx('jump',0.8);
      }
    }
  }

  if(pressed.atk){
    if(p.state==='idle' || p.state==='run' || p.state==='air'){
      if(p.onGround){ p.chain = 0; beginAttack('l1'); }
      else { p.airAtk++; beginAttack('air1'); }
    } else if(p.state==='attack'){
      if(p.t > p.atk.hs*0.6){ p.buffer='atk'; p.bufferT=0.28; }
    } else if(p.state==='dash' && p.t>0.07){
      p.state='idle';
      if(p.onGround) beginAttack('l1'); else { p.airAtk++; beginAttack('air1'); }
    }
  }

  if(pressed.heavy){
    if(p.state==='idle' || p.state==='run' || p.state==='air' || p.state==='dash' || p.state==='attack'){
      p.state='charge'; p.chargeT=0; p.t=0; p.atk=null;
    }
  }

  if(pressed.dash){
    if((p.state==='idle'||p.state==='run'||p.state==='air') && p.dashCd<=0){
      if(p.onGround){
        p.state='dash'; p.t=0; p.dashDur=0.24; p.iframe=0.19;
        p.vx = p.facing*680; p.dashCd=0.32;
        addSlash(p.x, p.y-42, -0.9, 0.9, 60, '#7fd0ff', 0.2, 10, p.facing);
        addDust(p.x, GY, 5); sfx('dash');
      } else if(p.airDash>0){
        p.airDash--; p.state='dash'; p.t=0; p.dashDur=0.19; p.iframe=0.16;
        p.vx = p.facing*600; p.vy = -40; p.dashCd=0.3;
        addRing(p.x, p.y-40, 6, 50, 0.3, '#7fd0ff');
        sfx('dash');
      }
    }
  }

  if(pressed.parry){
    if((p.state==='idle'||p.state==='run') && p.parryCd<=0){
      p.state='parry'; p.t=0; p.parryCd=0.7;
      p.vx *= 0.2;
      addRing(p.x+p.facing*18, p.y-48, 6, 54, 0.26, '#9fe0ff');
    }
  }

  /* ---- 物理 ---- */
  p.vy += GRAV*dt;
  if(p.vy > 1500) p.vy = 1500;
  p.y += p.vy*dt;
  p.x += p.vx*dt;

  if(p.y >= GY){
    if(!p.onGround){
      const impact = p.vy;
      p.onGround = true;
      p.jumps = 0; p.airDash = 1; p.airAtk = 0;
      if(impact > 520){ addDust(p.x, GY, 7); addShake(2.2); sfx('land',0.7); }
      if(p.state==='air') p.state='idle';
      if(p.state==='attack' && p.atk && p.atk.name==='air2'){ p.state='idle'; p.atk=null; }
    }
    p.y = GY; p.vy = 0;
  } else {
    p.onGround = false;
    if(p.state==='idle'||p.state==='run') p.state='air';
  }
  if(p.state==='air' && p.onGround) p.state='idle';

  p.x = clamp(p.x, 30, ARENA_W-30);
  if(p.state==='run') p.walkPh += dt*13;
}

/* ===================== 敌人更新 ===================== */
function updateEnemy(e, dt){
  if(e.dead){
    e.deadT += dt;
    e.vy += GRAV*dt;
    e.y += e.vy*dt; e.x += e.vx*dt;
    e.vx = approach(e.vx,0,600*dt);
    if(e.y>GY){ e.y=GY; e.vy = -e.vy*0.3; if(Math.abs(e.vy)<60) e.vy=0; }
    return;
  }
  e.t += dt;
  e.flash = Math.max(0, e.flash-dt);
  e.hitStun = Math.max(0, e.hitStun-dt);
  e.staggerT = Math.max(0, e.staggerT-dt);
  e.atkCd = Math.max(0, e.atkCd-dt);

  // 物理
  e.vy += GRAV*dt;
  e.y += e.vy*dt;
  if(e.y >= GY){
    if(!e.onGround && e.vy>300){ addDust(e.x, GY, 4); }
    e.y = GY; e.vy = 0; e.onGround = true;
  } else e.onGround = false;
  e.x += e.vx*dt;
  e.vx = approach(e.vx, 0, (e.onGround?1500:400)*dt);
  e.x = clamp(e.x, 40, ARENA_W-40);

  const p = player;
  const dx = p.x - e.x;
  const dist = Math.abs(dx);

  if(e.staggerT>0 || e.hitStun>0){
    if(e.state!=='recover'){ e.state='stagger'; }
    return;
  }
  if(e.state==='stagger'){ e.state='idle'; e.t=0; }

  if(p.state!=='dead' && dist>70) e.facing = dx>0?1:-1;

  const k = EKIND[e.kind];
  let spd = k.speed;
  if(e.kind==='dasher' && !hasPart(e,'leg')) spd *= 0.38;
  if(e.kind==='puppet' && !hasPart(e,'head')) spd *= 0.55;

  if(e.kind==='puppet'){
    const blind = !hasPart(e,'head');
    const arms = (hasPart(e,'armL')?1:0)+(hasPart(e,'armR')?1:0);
    switch(e.state){
      case 'spawn':
        if(e.t>0.4){ e.state='idle'; e.t=0; }
        break;
      case 'idle':
        if(p.state==='dead'){ break; }
        if(blind){ e.x += (Math.sin(e.t*1.6)>0?1:-1)*spd*0.5*dt; e.walkPh+=dt*6; break; }
        if(dist > 84){ e.x += e.facing*spd*dt; e.walkPh += dt*7; }
        else if(e.atkCd<=0){ e.state='windup'; e.t=0; e.hasHit=false; }
        break;
      case 'windup': {
        const wu = arms>=2 ? 0.52 : (arms===1 ? 0.6 : 0.46);
        if(dist>130 && arms>0){ e.state='idle'; e.atkCd=0.4; break; }
        if(e.t>=wu){ e.state='swing'; e.t=0; }
        break;
      }
      case 'swing': {
        if(arms>0){
          const dmg = arms>=2 ? 16 : 11;
          const hb = e.facing>0
            ? {x:e.x+6, y:e.y-78, w:78, h:64}
            : {x:e.x-84, y:e.y-78, w:78, h:64};
          if(!e.hasHit && e.t>0.03){
            if(checkPlayerHit(hb, dmg, e)) e.hasHit = true;
          }
          if(e.t>0.2){ e.state='recover'; e.t=0; e.atkCd=0.95; }
        } else {
          // 无臂：头槌
          const hb = e.facing>0
            ? {x:e.x+4, y:e.y-64, w:44, h:50}
            : {x:e.x-48, y:e.y-64, w:44, h:50};
          if(!e.hasHit && e.t>0.04){ if(checkPlayerHit(hb, 8, e)) e.hasHit=true; }
          if(e.t>0.18){ e.state='recover'; e.t=0; e.atkCd=1.3; }
        }
        break;
      }
      case 'recover':
        if(e.t>0.5){ e.state='idle'; e.t=0; }
        break;
    }
    if(e.state==='idle' && blind) e.walkPh += dt*4;
  }

  else { // dasher
    const blade = hasPart(e,'blade');
    const leg = hasPart(e,'leg');
    switch(e.state){
      case 'spawn':
        if(e.t>0.4){ e.state='idle'; e.t=0; }
        break;
      case 'idle':
        if(p.state==='dead'){ break; }
        if(!leg && dist<70 && e.atkCd<=0){
          e.state='windup'; e.t=0; e.hasHit=false; e.wuLen=0.4; break;
        }
        if(leg && dist>110 && dist<330){ e.x += e.facing*spd*dt; e.walkPh += dt*12; }
        else if(dist>=330){ e.x += e.facing*spd*dt; e.walkPh += dt*12; }
        else if(dist<=110 && e.atkCd<=0){
          e.state='windup'; e.t=0; e.hasHit=false; e.wuLen=0.42;
        }
        break;
      case 'windup':
        if(e.t>=(e.wuLen||0.42)){ e.state='lunge'; e.t=0; e.hasHit=false;
          e.vx = e.facing*(blade?620:380);
          addSlash(e.x, e.y-36, -0.6, 0.6, 40, '#ff7a7a', 0.18, 8, e.facing);
          sfx('dash',0.7);
        }
        break;
      case 'lunge': {
        const dmg = blade?14:7;
        const hb = e.facing>0
          ? {x:e.x-6, y:e.y-62, w:60, h:62}
          : {x:e.x-54, y:e.y-62, w:60, h:62};
        if(!e.hasHit){ if(checkPlayerHit(hb, dmg, e)) e.hasHit=true; }
        if(e.t>0.22){ e.state='recover'; e.t=0; e.atkCd = leg?1.15:1.6; }
        break;
      }
      case 'recover':
        if(e.t>0.42){ e.state='idle'; e.t=0; }
        break;
    }
  }

  // 简单分离
  for(let i=0;i<G.enemies.length;i++){
    const o = G.enemies[i];
    if(o===e || o.dead) continue;
    const d = e.x - o.x;
    if(Math.abs(d) < 34 && Math.abs(e.y-o.y)<40){
      const push = (34-Math.abs(d))*0.5;
      e.x += (d>=0?1:-1)*push*dt*8;
    }
  }
}

function checkPlayerHit(hb, dmg, e){
  const p = player;
  if(p.state==='dead') return false;
  if(p.iframe>0 || p.invuln>0) return false;
  const pr = playerRect();
  if(!rectRect(hb, pr)) return false;
  if(p.state==='parry' && p.t <= 0.21){
    doParry(e);
    e.state='recover'; e.t=0; e.atkCd=1.4; e.hasHit=true;
    return true;
  }
  damagePlayer(dmg, e.x);
  return true;
}

/* ===================== 特效更新 ===================== */
function updateFx(dt){
  for(let i=G.sparks.length-1;i>=0;i--){
    const s = G.sparks[i];
    s.life -= dt;
    s.vy += 1500*dt;
    s.x += s.vx*dt; s.y += s.vy*dt;
    s.vx *= 0.96;
    if(s.life<=0) G.sparks.splice(i,1);
  }
  for(let i=G.debris.length-1;i>=0;i--){
    const d = G.debris[i];
    d.life -= dt;
    d.vy += 1800*dt;
    d.x += d.vx*dt; d.y += d.vy*dt;
    d.rot += d.vr*dt;
    if(d.y > GY-3){
      d.y = GY-3;
      if(Math.abs(d.vy)>60){ d.vy = -d.vy*0.36; d.vx *= 0.72; d.vr *= 0.55; }
      else { d.vy=0; d.vx *= 0.86; d.vr *= 0.8; }
    }
    if(d.life<=0) G.debris.splice(i,1);
  }
  for(let i=G.slashes.length-1;i>=0;i--){
    G.slashes[i].life -= dt;
    if(G.slashes[i].life<=0) G.slashes.splice(i,1);
  }
  for(let i=G.texts.length-1;i>=0;i--){
    const t = G.texts[i];
    t.life -= dt; t.y += t.vy*dt; t.vy *= 0.9;
    if(t.life<=0) G.texts.splice(i,1);
  }
  for(let i=G.rings.length-1;i>=0;i--){
    G.rings[i].life -= dt;
    if(G.rings[i].life<=0) G.rings.splice(i,1);
  }
}

/* ===================== 波次更新 ===================== */
function updateWaves(dt){
  if(G.waveQueue.length>0){
    G.spawnTimer -= dt;
    if(G.spawnTimer<=0){
      G.spawnTimer = 0.34;
      spawnEnemy(G.waveQueue.shift());
    }
    return;
  }
  let alive = 0;
  for(let i=0;i<G.enemies.length;i++) if(!G.enemies[i].dead) alive++;
  if(alive===0){
    if(G.betweenWaves<=0){
      G.betweenWaves = 1.5;
    } else {
      G.betweenWaves -= dt;
      if(G.betweenWaves<=0) startWave();
    }
  }
}

/* ===================== 相机 ===================== */
function updateCamera(dt){
  const p = player;
  const target = clamp(p.x - VW/2 + p.facing*66, 0, ARENA_W - VW);
  const k = 1 - Math.pow(0.0009, dt);
  G.camera.x = lerp(G.camera.x, target, k);
  G.camera.shake = Math.max(0, G.camera.shake - dt*34);
  G.hurtVig = Math.max(0, G.hurtVig - dt*2.2);
}

/* ===================== 主循环 ===================== */
function update(dt){
  G.time += dt;
  updatePlayer(dt);
  for(let i=0;i<G.enemies.length;i++) updateEnemy(G.enemies[i], dt);
  updateFx(dt);
  updateWaves(dt);
  updateCamera(dt);
  for(let i=G.enemies.length-1;i>=0;i--){
    const e = G.enemies[i];
    if(e.dead && e.deadT>1.1) G.enemies.splice(i,1);
  }
  for(const k in pressed) delete pressed[k];
}

### 战斗系统与操作体验

- **连段与蓄力**：J 键触发三段斩击，每段衔接节奏紧凑，第三段带挑空效果；K 键长按蓄力，松开后释放重斩，蓄力越久伤害越高，命中后触发更长的顿帧和更强的震屏。
- **攻防博弈**：L 键冲刺带无敌帧，可用于规避敌人前兆明显的攻击；I 键弹反需要精准掌握时机，成功时打断敌人肢体并触发反击，失败则陷入小段硬直。
- **肢体破坏**：敌人分为木偶和镰足两种类型，各自带有可断裂的肢体。强力攻击或持续连段会击碎肢体，使敌人失去对应攻击手段或移动能力，断肢会飞散为物理碎块。
- **反馈与节奏**：命中顿帧时长根据攻击力度分级，配合轻微的震屏、刀光、火花和伤害数字。连击数会显示在屏幕右侧，波次推进时敌人持续刷新，死亡后按 R 或 Enter 即可快速重开。

**优化建议：** 如果您想调整战斗难度或节奏，可以修改脚本开头的玩家属性（如 `maxHp`、`SPEED`、`JUMPV`）和敌人属性（`EKIND.puppet.hp` 等），以及波次生成逻辑中的敌人数和类型。
