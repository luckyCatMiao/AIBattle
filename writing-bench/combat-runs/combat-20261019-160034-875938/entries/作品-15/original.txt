let actx=null,audioReady=false,muted=false,noiseBuf=null;
function ensureAudio(){
  if(audioReady) return;
  audioReady=true;
  try{
    const AC=window.AudioContext||window.webkitAudioContext;
    if(!AC) return;
    actx=new AC();
    if(actx.state==='suspended') actx.resume();
  }catch(e){ actx=null; }
}
function getNoise(){
  if(!noiseBuf&&actx){
    const len=Math.floor(actx.sampleRate*0.5);
    noiseBuf=actx.createBuffer(1,len,actx.sampleRate);
    const d=noiseBuf.getChannelData(0);
    for(let i=0;i<len;i++) d[i]=Math.random()*2-1;
  }
  return noiseBuf;
}
function nz(t,dur,f0,f1,gain,q){
  const src=actx.createBufferSource(); src.buffer=getNoise();
  const f=actx.createBiquadFilter(); f.type='bandpass'; f.Q.value=q||1.4;
  f.frequency.setValueAtTime(f0,t);
  f.frequency.exponentialRampToValueAtTime(Math.max(60,f1),t+dur);
  const g=actx.createGain();
  g.gain.setValueAtTime(gain,t);
  g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  src.connect(f); f.connect(g); g.connect(actx.destination);
  src.start(t); src.stop(t+dur+0.03);
}
function tn(t,f0,f1,dur,gain,type){
  const o=actx.createOscillator(); o.type=type||'sine';
  o.frequency.setValueAtTime(f0,t);
  o.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur);
  const g=actx.createGain();
  g.gain.setValueAtTime(gain,t);
  g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  o.connect(g); g.connect(actx.destination);
  o.start(t); o.stop(t+dur+0.03);
}

/* ============================ 玩家行为 ============================ */
function startAttack(name){
  const p=player, a=P_ATK[name];
  if(!a) return;
  const L=keys['KeyA']||keys['ArrowLeft'], R=keys['KeyD']||keys['ArrowRight'];
  if(R&&!L) p.facing=1; else if(L&&!R) p.facing=-1;
  p.state='attack'; p.atk=a; p.atkName=name; p.atkT=0;
  p.hitList=[]; p.bufferAtk=false; p.bufferHeavy=false;
  p.vx=p.facing*a.push;
  if(!p.onGround && p.vy>0) p.vy*=-0.25;
  sfx(name==='heavy'?'swingHeavy':'swing');
}

function updateAttack(dt){
  const p=player, a=p.atk;
  if(!a){ p.state='free'; return; }
  p.atkT+=dt;
  const t=p.atkT;
  const accel=a.startup+a.active;

  if(t>accel) p.vx*=Math.pow(0.02,dt);

  if(p.bufferAtk && t>a.startup+a.active*0.45){
    p.bufferAtk=false;
    if(a.next){ startAttack(a.next); return; }
  }
  if(p.bufferHeavy && t>a.startup+a.active*0.45){
    p.bufferHeavy=false;
    startAttack('heavy'); return;
  }

  if(t>=a.startup && t<=a.startup+a.active) doPlayerHit(a);

  if(t>=a.startup+a.active+a.recover){
    p.state='free'; p.atk=null; p.atkName=null;
  }
}

function doPlayerHit(a){
  const p=player, f=p.facing;
  const bx = f>0 ? p.x+4 : p.x-4-a.reach;
  const by = p.y+a.y0;
  const bw = a.reach, bh = a.h;
  for(let i=0;i<enemies.length;i++){
    const e=enemies[i];
    if(e.dead) continue;
    if(p.hitList.indexOf(e)>=0) continue;
    const ex=e.x-26, ey=e.y-78, ew=52, eh=78;
    if(bx<ex+ew && bx+bw>ex && by<ey+eh && by+bh>ey){
      p.hitList.push(e);
      hitEnemy(e,a);
    }
  }
}

function hitEnemy(e,a){
  const p=player, dir=p.facing;
  const px=e.x+dir*10, py=e.y-60;

  // 选择肢体：优先朝向玩家那侧
  let limb=null;
  const wantSide=Math.sign(p.x-e.x)||1;
  for(const l of e.limbs){ if(!l.broken && Math.sign(l.side)===wantSide){ limb=l; break; } }
  if(!limb){ for(const l of e.limbs){ if(!l.broken){ limb=l; break; } } }

  e.hp-=a.dmg;
  e.flash=0.14;
  e.kbVX=dir*a.kb;
  e.staggerT=Math.max(e.staggerT,a.stagger||0.16);
  if(a.launch && e.onGround){ e.vy=a.launch; e.onGround=false; }

  if(limb){
    limb.hp-=a.limb;
    if(limb.hp<=0) breakLimb(e,limb);
  }

  // 顿帧 / 震屏
  state.hitStop=Math.max(state.hitStop,a.hs);
  state.shake=Math.max(state.shake,a.shake||4);

  // 特效
  spawnSpark(px,py,a.type==='heavy'?18:10, a.type==='heavy'?'#ffd08a':'#a9ecff', 460);
  spawnSpark(px,py,6,'#ffffff',260);
  slashes.push({x:px,y:py,ang:rand(-0.9,0.9)+(dir>0?0:PI),len:a.type==='heavy'?70:40,
    arc:1.6,life:0.16,max:0.16,color:a.type==='heavy'?'#ffc36b':'#bff4ff'});
  addDamageText(px,py-16,a.dmg+'',a.type==='heavy'?'#ffd479':'#9fe8ff');

  // 能量 / 连击
  p.energy=Math.min(100,p.energy+(a.type==='heavy'?11:7));
  state.hitCount++;
  addCombo();
  sfx(a.type==='heavy'?'hitHeavy':'hit');

  if(e.hp<=0) killEnemy(e);
}

function addCombo(){
  state.combo++;
  state.comboTimer=1.25;
  if(state.combo>state.maxCombo) state.maxCombo=state.combo;
}

function breakLimb(e,limb){
  limb.broken=true;
  state.limbBreakCount++;
  const wx=e.x+limb.side*12*e.facing, wy=e.y-62;
  debris.push({
    x:wx,y:wy,
    vx:player.facing*rand(160,340)+rand(-70,70),
    vy:rand(-560,-280),
    rot:rand(0,6.28), vr:rand(-16,16),
    len:rand(24,34), w:9, life:2.8, max:2.8, side:limb.side
  });
  spawnSpark(wx,wy,20,'#ffb35c',560);
  spawnSpark(wx,wy,10,'#ff5a3c',420);
  state.hitStop=Math.max(state.hitStop,0.11);
  state.shake=Math.max(state.shake,11);
  e.staggerT=Math.max(e.staggerT,0.5);
  e.kbVX+=player.facing*140;
  sfx('break');
}

  e.dead=true;
  state.kills++;

  state.hitStop=Math.max(state.hitStop,0.07);

      len:rand(14,26),w:rand(6,10),life:1.8,max:1.8,side:1});
  }
  sfx('die');
  const idx=enemies.indexOf(e);
  if(idx>=0) enemies.splice(idx,1);
}

  p.iframe=Math.max(0,p.iframe-dt);
  p.dashCd=Math.max(0,p.dashCd-dt);
  p.hurtT=Math.max(0,p.hurtT-dt);
  p.flash=Math.max(0,p.flash-dt);

  const L=keys['KeyA']||keys['ArrowLeft'];
  const R=keys['KeyD']||keys['ArrowRight'];
  const mx=(R?1:0)-(L?1:0);

  if(p.jumpBuf>0) p.jumpBuf-=dt;
  if(pressed['Space']&&p.state!=='dead') p.jumpBuf=0.13;
  if(p.onGround) p.coyote=0.09; else p.coyote=Math.max(0,p.coyote-dt);

  if(p.state==='dead'){
    p.vx*=Math.pow(0.05,dt);
  } else if(p.state==='dash'){
    p.dashT-=dt;
    p.vx=p.dashDir*840;
    if(p.dashT<=0){ p.state='free'; p.vx*=0.35; }
  } else if(p.state==='attack'){
    if(pressed['KeyJ']) p.bufferAtk=true;
    if(pressed['KeyK']) p.bufferHeavy=true;
    if(pressed['KeyL']) tryDash(mx);
    updateAttack(dt);
  } else {
    // 自由移动
    if(mx!==0){
      p.vx+=mx*4200*dt;
      p.vx=clamp(p.vx,-340,340);
      p.facing=mx;
    } else {
      p.vx*=Math.pow(p.onGround?0.0002:0.22,dt);
    }
    // 跳跃
    if(p.jumpBuf>0&&p.coyote>0){
      p.vy=-860; p.onGround=false; p.coyote=0; p.jumpBuf=0;
      spawnSpark(p.x,GROUND_Y,5,'#8fd4ff',180);
      sfx('jump');
    }
    if(!keys['Space']&&p.vy<-320) p.vy=-320;

    if(pressed['KeyJ']) startAttack('l1');
    else if(pressed['KeyK']) startAttack('heavy');
    else if(pressed['KeyL']) tryDash(mx);
    else if(pressed['KeyI']) tryBurst();
  }

  // 物理
  p.vy+=GRAV*dt;
  if(p.vy>1500) p.vy=1500;
  p.x+=p.vx*dt;
  p.y+=p.vy*dt;
  if(p.y>=GROUND_Y){
    if(!p.onGround&&p.vy>200) spawnSpark(p.x,GROUND_Y,4,'#8fd4ff',140);
    p.y=GROUND_Y; p.vy=0; p.onGround=true;
  } else {
    p.onGround=false;
  }
  p.x=clamp(p.x,26,ARENA_W-26);
}

/* ============================ 敌人 ============================ */
function makeEnemy(type,x){
  const e={
    type:type,x:x,y:GROUND_Y,vx:0,vy:0,aiVX:0,kbVX:0,facing:-1,
    hp:type==='grunt'?62:42,maxHp:type==='grunt'?62:42,
    onGround:true,mode:'chase',timer:0,cd:rand(0.4,1.1),
    staggerT:0,flash:0,hitDone:false,animT:rand(0,10),dead:false,
    limbs:[
      {side:-1,hp:14,max:14,broken:false},
      {side:1,hp:14,max:14,broken:false}
    ]
  };
  return e;
}

function updateEnemies(dt){
  for(let i=0;i<enemies.length;i++){
    const e=enemies[i];
    if(e.dead) continue;
    e.animT+=dt;
    e.flash=Math.max(0,e.flash-dt);
    e.staggerT=Math.max(0,e.staggerT-dt);
    e.cd=Math.max(0,e.cd-dt);

    const dx=player.x-e.x;
    if(Math.abs(dx)>4) e.facing=Math.sign(dx);

    if(!e.onGround){
      e.aiVX*=0.98;
    } else if(e.staggerT>0){
      e.aiVX=0;
    } else {
      const dist=Math.abs(dx);
      const dir=Math.sign(dx)||1;
      const arms=e.limbs.filter(function(l){return !l.broken;}).length;
      if(e.type==='grunt') gruntAI(e,dt,dist,dir,arms);
      else gunnerAI(e,dt,dist,dir,arms);
    }

    // 击退衰减
    e.kbVX*=Math.pow(e.onGround?0.0008:0.25,dt);

    // 物理
    e.vy+=GRAV*dt;
    e.x+=(e.aiVX+e.kbVX)*dt;
    e.y+=e.vy*dt;
    if(e.y>=GROUND_Y){
      if(!e.onGround){
        e.y=GROUND_Y; e.vy=0; e.onGround=true;
        spawnSpark(e.x,GROUND_Y,8,'#b07bff',260);
        state.shake=Math.max(state.shake,3);
        e.staggerT=Math.max(e.staggerT,0.3);
      }
      e.y=GROUND_Y; e.vy=0;
    } else {
      e.onGround=false;
    }
    e.x=clamp(e.x,26,ARENA_W-26);

    // 近身接触微弱推挤，避免重叠
    for(let j=i+1;j<enemies.length;j++){
      const o=enemies[j];
      const d=o.x-e.x;
      if(Math.abs(d)<40){
        const push=(40-Math.abs(d))*0.5*Math.sign(d||1);
        e.x-=push*0.02; o.x+=push*0.02;
      }
    }
  }
}

function gruntAI(e,dt,dist,dir,arms){
  switch(e.mode){
    case 'chase':
      if(dist>84){
        e.aiVX=dir*(arms>0?150:200);
      } else {
        e.aiVX=0;
        if(e.cd<=0){ e.mode='windup'; e.timer=arms>0?0.44:0.66; }
      }
      break;
    case 'windup':
      e.aiVX=0; e.timer-=dt;
      if(dist>170){ e.mode='chase'; e.cd=0.3; break; }
      if(e.timer<=0){ e.mode='swing'; e.timer=0.19; e.hitDone=false; }
      break;
    case 'swing':
      e.aiVX=dir*185; e.timer-=dt;
      if(!e.hitDone) enemyStrike(e,dir,86,arms);
      if(e.timer<=0){ e.mode='recover'; e.timer=arms>0?0.42:0.72; e.cd=rand(0.85,1.4); }
      break;
    case 'recover':
      e.aiVX=0; e.timer-=dt;
      if(e.timer<=0) e.mode='chase';
      break;
  }
}

function gunnerAI(e,dt,dist,dir,arms){
  switch(e.mode){
    case 'chase':
      if(arms===0){ e.mode='flee'; break; }
      if(dist<210) e.aiVX=-dir*140;
      else if(dist>350) e.aiVX=dir*140;
      else e.aiVX=0;
      if(e.cd<=0&&dist>100){ e.mode='windup'; e.timer=0.62; }
      break;
    case 'windup':
      e.aiVX=0; e.timer-=dt;
      if(e.timer<=0){ e.mode='shoot'; e.timer=0.12; e.hitDone=false; }
      break;
    case 'shoot':
      e.aiVX=0; e.timer-=dt;
      if(!e.hitDone){ e.hitDone=true; fireProjectile(e); }
      if(e.timer<=0){ e.mode='recover'; e.timer=0.5; e.cd=rand(1.5,2.1); }
      break;
    case 'recover':
      e.aiVX=0; e.timer-=dt;
      if(e.timer<=0) e.mode='chase';
      break;
    case 'flee':
      e.aiVX=-dir*95;
      if(dist>340) e.aiVX=0;
      break;
  }
}

function enemyStrike(e,dir,reach,arms){
  const p=player;
  if(p.state==='dead'||p.iframe>0) return;
  const bx=dir>0?e.x+8:e.x-8-reach;
  const by=e.y-80;
  const px=p.x-16, py=p.y-68, pw=32, ph=68;
  if(bx<px+pw&&bx+reach>px&&by<py+ph&&by+72>py){
    e.hitDone=true;
    hurtPlayer(arms>0?11:6,dir);
  }
}

function fireProjectile(e){
  const p=player;
  const dx=p.x-e.x, dy=(p.y-42)-(e.y-62);
  const d=Math.hypot(dx,dy)||1;
  const sp=430;
  projectiles.push({x:e.x+e.facing*22,y:e.y-62,vx:dx/d*sp,vy:dy/d*sp,r:9,life:3.2});
  sfx('shot');
}

function updateProjectiles(dt){
  const p=player;
  for(let i=projectiles.length-1;i>=0;i--){
    const pr=projectiles[i];
    pr.x+=pr.vx*dt; pr.y+=pr.vy*dt; pr.life-=dt;
    if(pr.life<=0||pr.x<-100||pr.x>ARENA_W+100||pr.y>GROUND_Y+30){
      projectiles.splice(i,1); continue;
    }
    const px=p.x-16, py=p.y-68;
    if(p.state!=='dead'&&p.iframe<=0&&
       pr.x>px&&pr.x<px+32&&pr.y>py&&pr.y<py+68){
      hurtPlayer(9,Math.sign(pr.vx)||1);
      spawnSpark(pr.x,pr.y,10,'#8ef0ff',300);
      projectiles.splice(i,1);
    }
  }
}

/* ============================ 粒子 / 碎片 ============================ */
function spawnSpark(x,y,n,color,spd){
  for(let i=0;i<n;i++){
    const a=rand(0,PI*2), s=rand(spd*0.25,spd);
    particles.push({
      x:x,y:y,vx:Math.cos(a)*s,vy:Math.sin(a)*s-60,
      life:rand(0.18,0.46),max:0.46,color:color,
      size:rand(1.6,3.8),grav:1100
    });
  }
}
function updateParticles(dt){
  for(let i=particles.length-1;i>=0;i--){
    const p=particles[i];
    p.x+=p.vx*dt; p.y+=p.vy*dt;
    p.vy+=p.grav*dt;
    p.vx*=Math.pow(0.15,dt);
    p.life-=dt;
    if(p.life<=0) particles.splice(i,1);
  }
  for(let i=slashes.length-1;i>=0;i--){
    slashes[i].life-=dt;
    if(slashes[i].life<=0) slashes.splice(i,1);
  }
}
function updateDebris(dt){
  for(let i=debris.length-1;i>=0;i--){
    const d=debris[i];
    d.vy+=GRAV*dt;
    d.x+=d.vx*dt; d.y+=d.vy*dt;
    d.rot+=d.vr*dt;
    if(d.y>GROUND_Y-4){
      d.y=GROUND_Y-4; d.vy*=-0.42; d.vx*=0.65; d.vr*=0.6;
      if(Math.abs(d.vy)<60) d.vy=0;
    }
    d.life-=dt;
    if(d.life<=0) debris.splice(i,1);
  }
}

const dmgTexts=[];
function addDamageText(x,y,txt,color){
  dmgTexts.push({x:x,y:y,txt:txt,color:color,life:0.55,max:0.55,vy:-90});
}
function updateDmgTexts(dt){
  for(let i=dmgTexts.length-1;i>=0;i--){
    const d=dmgTexts[i];
    d.y+=d.vy*dt; d.vy*=Math.pow(0.1,dt);
    d.life-=dt;
    if(d.life<=0) dmgTexts.splice(i,1);
  }
}

/* ============================ 波次 ============================ */
function startWave(){
  state.wave++;
  state.waveTimer=0;
  const w=state.wave;
  const grunts=Math.min(1+Math.ceil(w*0.7),4);
  const gunners=w>=2?Math.min(Math.floor((w-1)*0.8),3):0;
  const list=[];
  for(let i=0;i<grunts;i++) list.push('grunt');
  for(let i=0;i<gunners;i++) list.push('gunner');
  // 打乱
  for(let i=list.length-1;i>0;i--){
    const j=Math.floor(Math.random()*(i+1));
    const tmp=list[i]; list[i]=list[j]; list[j]=tmp;
  }
  let t=0;
  for(const type of list){
    const leftSide = Math.abs(player.x-220) > Math.abs(player.x-(ARENA_W-220));
    const x = leftSide ? rand(120,340) : rand(ARENA_W-340,ARENA_W-120);
    state.spawnQueue.push({type:type,x:x,t:t});
    t+=0.28;
  }
  document.getElementById('waveLabel').textContent='WAVE '+w;
  sfx('wave');
}

function updateWaves(dt){
  if(state.spawnQueue.length>0){
    const s=state.spawnQueue[0];
    s.t-=dt;
    if(s.t<=0){
      state.spawnQueue.shift();
      const e=makeEnemy(s.type,s.x);
      enemies.push(e);
      spawnSpark(s.x,GROUND_Y-40,18,s.type==='grunt'?'#ff7a4c':'#6ee7ff',420);
    }
    return;
  }
  if(enemies.length===0){
    state.waveTimer+=dt;
    if(state.waveTimer>0.85) startWave();
  }
}

/* ============================ 摄像机 ============================ */
function updateCamera(dt){
  const p=player;
  const target=p.x-viewW*0.5+p.facing*70;
  const k=1-Math.pow(0.0009,dt);
  state.camX=lerp(state.camX,target,k);
  const maxCam=ARENA_W-viewW;
  if(maxCam<=0) state.camX=maxCam/2;
  else state.camX=clamp(state.camX,0,maxCam);

  if(state.shake>0.1){
    state.shakeX=(Math.random()-0.5)*state.shake;
    state.shakeY=(Math.random()-0.5)*state.shake;
  } else {
    state.shake=0; state.shakeX=0; state.shakeY=0;
  }
}

/* ============================ 主更新 ============================ */
function update(dt){
  state.t+=dt;
  if(state.shake>0) state.shake=Math.max(0,state.shake-dt*46);
  state.flash=Math.max(0,state.flash-dt*2.4);

  if(state.hitStop>0){
    state.hitStop-=dt;
    return;
  }

  updatePlayer(dt);
  if(state.playing){
    updateEnemies(dt);
    updateProjectiles(dt);
    updateWaves(dt);
  }
  updateParticles(dt);
  updateDebris(dt);
  updateDmgTexts(dt);
  updateCamera(dt);

  if(state.comboTimer>0){
    state.comboTimer-=dt;
    if(state.comboTimer<=0) state.combo=0;
  }
}

  const end=Math.ceil((state.camX*par+viewW)/spacing)+1;

    const sx=i*spacing-state.camX*par;
    const r=hash(i*13.37+seed);
    const h=hMin+r*(hMax-hMin);
    const w=spacing*(0.55+hash(i*7.1+seed)*0.32);

    if(hash(i*3.7+seed)>0.6){
      ctx.fillRect(sx+w*0.45,baseY-h-16,3,16);
    }
  }
}

/* ---------- 玩家绘制 ---------- */
function drawPlayer(){
  const p=player;
  const f=p.facing, t=p.animT;
  const flash=p.flash>0;
  const body=flash?'#fff0ec':'#e6f6ff';
  const edge=flash?'#ff6a4d':'#49c9ff';

  ctx.save();
  ctx.translate(p.x,p.y);
  ctx.scale(f,1);

  // 影子
  ctx.globalAlpha=0.35;
  ctx.fillStyle='#000';
  ctx.beginPath(); ctx.ellipse(0,2,20,6,0,0,PI*2); ctx.fill();
  ctx.globalAlpha=1;

  const hipY=-36, neckY=-70;

  // 移动摆动
  let lp=0;
  if(p.onGround&&Math.abs(p.vx)>25) lp=Math.sin(t*15)*clamp(Math.abs(p.vx)/320,0,1);
  let airTuck=0;
  if(!p.onGround) airTuck=clamp(-p.vy/700,-1,1);

  if(p.state==='dead'){
    // 倒地
    ctx.strokeStyle=body; ctx.lineWidth=9; ctx.lineCap='round';
    ctx.beginPath(); ctx.moveTo(-8,-12); ctx.lineTo(20,-14); ctx.stroke();
    ctx.beginPath(); ctx.arc(26,-16,8,0,PI*2); ctx.fillStyle=body; ctx.fill();
    ctx.restore();
    return;
  }

  // 红围巾
  ctx.strokeStyle=flash?'#ff9c8a':'#ff4d6d';
  ctx.lineWidth=5; ctx.lineJoin='round'; ctx.lineCap='round';
  ctx.beginPath();
  ctx.moveTo(2,neckY-4);
  for(let i=1;i<=4;i++){
    const sx=2-i*8;
    const sy=neckY-4+Math.sin(t*7-i*0.9)*4+i*2.4+(p.vy>200?4:0);
    ctx.lineTo(sx,sy);
  }
  ctx.stroke();

  // 后腿
  ctx.strokeStyle=edge; ctx.lineWidth=6; ctx.lineCap='round';
  ctx.beginPath();
  ctx.moveTo(-2,hipY);
  ctx.lineTo(-8-lp*9+(airTuck*6),hipY+18);
  ctx.lineTo(-11-lp*15,0);
  ctx.stroke();

  // 前腿
  ctx.strokeStyle=body;
  ctx.beginPath();
  ctx.moveTo(2,hipY);
  ctx.lineTo(8+lp*9,hipY+18);
  ctx.lineTo(11+lp*15,0);
  ctx.stroke();

  // 躯干
  ctx.strokeStyle=body; ctx.lineWidth=13;
  ctx.beginPath(); ctx.moveTo(0,hipY); ctx.lineTo(3,neckY); ctx.stroke();
  // 躯干亮边
  ctx.strokeStyle=edge; ctx.lineWidth=3;
  ctx.beginPath(); ctx.moveTo(1,hipY); ctx.lineTo(4,neckY); ctx.stroke();

  // 头
  ctx.fillStyle=body;
  ctx.beginPath(); ctx.arc(5,neckY-9,8,0,PI*2); ctx.fill();
  ctx.fillStyle=edge;
  ctx.fillRect(8,neckY-12,7,3);

  // 持剑臂
  const a=p.atk;
  let th=1.55, armR=27, swordLen=52;
  if(p.state==='attack'&&a){
    const th0=a.theta0, th1=a.theta1;
    if(p.atkT<a.startup){
      const c=p.atkT/a.startup;
      th=lerp(th0+0.4,th0,c);
    } else if(p.atkT<a.startup+a.active){
      const c=(p.atkT-a.startup)/a.active;
      th=lerp(th0,th1,c*c*(3-2*c));
    } else {
      th=th1;
    }
    if(a.type==='heavy'){ armR=30; swordLen=64; }
  }
  const shx=3, shy=neckY+6;
  const hx=shx+Math.cos(th)*armR;
  const hy=shy+Math.sin(th)*armR;
  ctx.strokeStyle=body; ctx.lineWidth=7; ctx.lineCap='round';
  ctx.beginPath(); ctx.moveTo(shx,shy); ctx.lineTo(hx,hy); ctx.stroke();

  // 刀
  const tipX=hx+Math.cos(th)*swordLen;
  const tipY=hy+Math.sin(th)*swordLen;
  ctx.strokeStyle='rgba(120,230,255,0.28)'; ctx.lineWidth=9;
  ctx.beginPath(); ctx.moveTo(hx,hy); ctx.lineTo(tipX,tipY); ctx.stroke();
  ctx.strokeStyle=flash?'#ffffff':'#bff4ff'; ctx.lineWidth=4;
  ctx.beginPath(); ctx.moveTo(hx,hy); ctx.lineTo(tipX,tipY); ctx.stroke();
  ctx.strokeStyle='#ffffff'; ctx.lineWidth=1.6;
  ctx.beginPath(); ctx.moveTo(hx,hy); ctx.lineTo(tipX,tipY); ctx.stroke();

  ctx.restore();

  // 刀光轨迹（世界坐标）
  const hwx=p.x+f*hx, hwy=p.y+hy;
  const twx=p.x+f*tipX, twy=p.y+tipY;
  if(p.state==='attack'||p.state==='dash'){
    p.trail.push({hx:hwx,hy:hwy,tx:twx,ty:twy,life:0.14});
  }
  if(p.trail.length>12) p.trail.shift();
  ctx.lineCap='round';
  for(let i=0;i<p.trail.length;i++){
    const tr=p.trail[i];
    const al=(i/p.trail.length)*0.5;
    ctx.strokeStyle='rgba(150,235,255,'+al.toFixed(3)+')';
    ctx.lineWidth=1+ (i/p.trail.length)*5;
    ctx.beginPath(); ctx.moveTo(tr.hx,tr.hy); ctx.lineTo(tr.tx,tr.ty); ctx.stroke();
  }
  for(let i=p.trail.length-1;i>=0;i--){
    p.trail[i].life-=1/60;
    if(p.trail[i].life<=0) p.trail.splice(i,1);
  }
}

/* ---------- 敌人绘制 ---------- */
function drawEnemies(){
  for(const e of enemies){
    ctx.save();
    ctx.translate(e.x,e.y);
    ctx.scale(e.facing,1);

    const flash=e.flash>0;
    const isG=e.type==='grunt';
    const body=flash?'#ffffff':(isG?'#31284a':'#252a44');
    const edge=flash?'#ffffff':(isG?'#ff6a3c':'#6ee7ff');
    const dark=flash?'#ffffff':'#151022';

    // 影子
    ctx.globalAlpha=0.3; ctx.fillStyle='#000';
    ctx.beginPath(); ctx.ellipse(0,2,22,6,0,0,PI*2); ctx.fill();
    ctx.globalAlpha=1;

    const hipY=-38, neckY=-66;
    const sway=Math.sin(e.animT*6)*2;

    // 腿
    ctx.lineCap='round';
    ctx.strokeStyle=dark; ctx.lineWidth=8;
    ctx.beginPath(); ctx.moveTo(-4,hipY); ctx.lineTo(-10,hipY+18); ctx.lineTo(-12,0); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(4,hipY); ctx.lineTo(10,hipY+18); ctx.lineTo(12,0); ctx.stroke();

    // 躯干
    ctx.fillStyle=body;
    ctx.beginPath();
    ctx.moveTo(-13,hipY+2);
    ctx.lineTo(-10,neckY);
    ctx.lineTo(10,neckY);
    ctx.lineTo(13,hipY+2);
    ctx.closePath();
    ctx.fill();
    // 核心
    ctx.fillStyle=edge;
    ctx.globalAlpha=0.85;
    ctx.beginPath(); ctx.arc(0,neckY+16,4.5,0,PI*2); ctx.fill();
    ctx.globalAlpha=1;

    // 头
    ctx.fillStyle=dark;
    ctx.beginPath();
    ctx.moveTo(-8,neckY-4); ctx.lineTo(8,neckY-4);
    ctx.lineTo(6,neckY-18); ctx.lineTo(-6,neckY-18);
    ctx.closePath(); ctx.fill();
    // 眼
    ctx.fillStyle=edge;
    ctx.fillRect(1,neckY-13,7,3);

    // 手臂
    for(const l of e.limbs){
      const shx=l.side*11, shy=neckY+4;
      if(l.broken){
        ctx.strokeStyle='#ff9b3c'; ctx.lineWidth=5;
        ctx.beginPath(); ctx.moveTo(shx,shy); ctx.lineTo(shx*1.3,shy+8); ctx.stroke();
        ctx.fillStyle='#ffc46b';
        ctx.beginPath(); ctx.arc(shx*1.3,shy+9,2.5,0,PI*2); ctx.fill();
        continue;
      }
      let th=1.9;
      if(e.mode==='windup') th = (isG?2.75:0.15);
      else if(e.mode==='swing') th=0.65;
      else if(e.mode==='shoot') th=0.0;
      else if(e.mode==='recover') th=1.2;
      const L=isG?26:24;
      const hx=shx+Math.cos(th)*L;
      const hy=shy+Math.sin(th)*L;
      ctx.strokeStyle=body; ctx.lineWidth=6;
      ctx.beginPath(); ctx.moveTo(shx,shy); ctx.lineTo(hx,hy); ctx.stroke();
      ctx.fillStyle=dark;
      ctx.beginPath(); ctx.arc(hx,hy,5,0,PI*2); ctx.fill();
      if(!isG){
        ctx.strokeStyle=edge; ctx.lineWidth=3;
        ctx.beginPath(); ctx.moveTo(hx,hy); ctx.lineTo(hx+Math.cos(th)*10,hy+Math.sin(th)*10); ctx.stroke();
      }
      // 蓄力预警光
      if(e.mode==='windup'){
        const pulse=0.5+0.5*Math.sin(e.animT*24);
        ctx.fillStyle='rgba(255,90,60,'+(0.35+pulse*0.55)+')';
        ctx.beginPath(); ctx.arc(hx,hy,7+pulse*3,0,PI*2); ctx.fill();
      }
    }

    // 重击预警环
    if(e.mode==='windup'){
      const pulse=0.5+0.5*Math.sin(e.animT*22);
      ctx.strokeStyle='rgba(255,110,60,'+(0.25+pulse*0.5)+')';
      ctx.lineWidth=2;
      ctx.beginPath(); ctx.arc(0,neckY+10,30+pulse*8,0,PI*2); ctx.stroke();
    }

    ctx.restore();
  }
}

/* ---------- 其它绘制 ---------- */
function drawDebris(){
  for(const d of debris){
    const al=clamp(d.life/d.max,0,1);
    ctx.save();
    ctx.globalAlpha=al;
    ctx.translate(d.x,d.y);
    ctx.rotate(d.rot);
    ctx.fillStyle='#3a2e54';
    ctx.fillRect(-d.w/2,-d.len/2,d.w,d.len);
    ctx.fillStyle='#ff9b3c';
    ctx.fillRect(-d.w/2,-d.len/2,d.w,3);
    ctx.fillStyle='#ffd08a';
    ctx.fillRect(-2,-d.len/2-2,4,4);
    ctx.restore();
  }
}

function drawParticles(){
  for(const p of particles){
    const al=clamp(p.life/p.max,0,1);
    ctx.globalAlpha=al;
    ctx.fillStyle=p.color;
    const s=p.size*(0.5+al*0.5);
    ctx.fillRect(p.x-s/2,p.y-s/2,s,s);
  }
  ctx.globalAlpha=1;
}

function drawSlashes(){
  for(const s of slashes){
    const al=clamp(s.life/s.max,0,1);
    ctx.save();
    ctx.translate(s.x,s.y);
    ctx.rotate(s.ang);
    ctx.globalAlpha=al*0.85;
    ctx.strokeStyle=s.color;
    ctx.lineWidth=3+al*4;
    ctx.beginPath();
    ctx.arc(0,0,s.len, -s.arc*0.5, s.arc*0.5);
    ctx.stroke();
    ctx.globalAlpha=al*0.35;
    ctx.lineWidth=10;
    ctx.stroke();
    ctx.restore();
  }
  ctx.globalAlpha=1;
}

function drawProjectiles(){
  for(const pr of projectiles){
    const g=ctx.createRadialGradient(pr.x,pr.y,1,pr.x,pr.y,pr.r*2.2);
    g.addColorStop(0,'#e8ffff');
    g.addColorStop(0.4,'#6ee7ff');
    g.addColorStop(1,'#2f8fff00');
    ctx.fillStyle=g;
    ctx.beginPath(); ctx.arc(pr.x,pr.y,pr.r*2.2,0,PI*2); ctx.fill();
    ctx.fillStyle='#ffffff';
    ctx.beginPath(); ctx.arc(pr.x,pr.y,pr.r*0.45,0,PI*2); ctx.fill();
  }
}

function drawDmgTexts(){

}

/* ============================ 主循环 ============================ */
let last=performance.now(), acc=0;

function loop(now){
  requestAnimationFrame(loop);
  let dt=(now-last)/1000;
  last=now;
  if(dt>0.1) dt=0.1;
  acc+=dt;
  let steps=0;
  while(acc>=STEP&&steps<6){
    update(STEP);
    acc-=STEP;
    steps++;
  }
  if(steps>0){ for(const k in pressed) delete pressed[k]; }
  if(acc>STEP*6) acc=0;

  // 全局按键
  if(pressed['Enter']){
    if(!state.playing) startGame();
  }
  if(pressed['KeyR']&&state.started){
    startGame();
  }

  render();
  updateHud();
}
