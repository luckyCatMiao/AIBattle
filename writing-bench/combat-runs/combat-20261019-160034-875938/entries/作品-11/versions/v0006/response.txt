function initAudio(){

function nz(dur,type,f0,f1,vol,q){
  if(!AC)return;const t=AC.currentTime;
  const s=AC.createBufferSource();s.buffer=noiseBuf;s.loop=true;
  const f=AC.createBiquadFilter();f.type=type;f.Q.value=q||1;
  f.frequency.setValueAtTime(Math.max(20,f0),t);
  f.frequency.exponentialRampToValueAtTime(Math.max(20,f1),t+dur);
  const g=AC.createGain();g.gain.setValueAtTime(vol,t);
  g.gain.exponentialRampToValueAtTime(.001,t+dur);
  s.connect(f);f.connect(g);g.connect(master);s.start(t);s.stop(t+dur+.05);
}

}

    const ph1=d.st+d.ac,end=d.st+d.ac+d.rc;
    if(P.t<d.st)P.vx*=Math.exp(-8*dt);
    else if(P.t<ph1){
      if(P.t-dt<d.st){
        P.vx=P.face*d.step;
        sfx.swing(P.atk.name==='rl'?1.35:1+(d.dmg>12?.25:0));
        arcs.push({fol:1,a0:d.a0,a1:d.a1,r0:22,r1:d.rng+8,life:.16,max:.16,face:P.face,col:'190,235,255',cx:0,cy:-46});
        if(d.shock){ringFx(P.x,GY,10,130,.28);sfx.shock();}
      }
      doHits(d);P.vx*=Math.exp(-5*dt);
    }else{
      P.vx*=Math.exp(-6*dt);
      if(d.nxt&&P.jBuf>0&&P.t>ph1+d.rc*.2){P.jBuf=0;startAtk(NAMED[d.nxt]);}
      else if(P.t>=end)P.state=P.onGround?'idle':'air';
    }
    if(P.lBuf>0&&P.t>ph1){P.lBuf=0;if(P.state==='attack')startDash(mv);}
  }
  else{ // idle / run / air
    const acc=P.onGround?14:8;
    P.vx+=(mv*260-P.vx)*Math.min(1,acc*dt);
    if(mv)P.face=mv;
    P.state=P.onGround?(Math.abs(P.vx)>30?'run':'idle'):'air';
    if(P.spBuf>0){
      if(P.onGround||P.coyote>0){P.vy=-620;P.onGround=false;P.coyote=0;P.jumps=1;P.spBuf=0;P.squash=-.3;sfx.jump(0);dust(P.x,GY,4,0);}
      else if(P.jumps<2){P.vy=-540;P.jumps=2;P.spBuf=0;P.squash=-.25;sfx.jump(1);ringFx(P.x,P.y+8,4,26,.16);}
    }
    if(P.jBuf>0){
      P.jBuf=0;
      startAtk(P.onGround?NAMED.a1:NAMED.ai1);
    }
    if(P.iBuf>0&&P.onGround){P.iBuf=0;startAtk(NAMED.la);}
    if(P.lBuf>0){P.lBuf=0;startDash(mv);}
  }
  // 物理
  let g=1500;
  if(st==='attack'&&!P.onGround&&P.atk.d.air)g=560;
  if(st==='plunge'){P.vy=980;P.vx=0;}
  else if(st!=='dash')P.vy+=g*dt;
  const wasG=P.onGround,pvy=P.vy,pst=P.state;
  P.x+=P.vx*dt;P.y+=P.vy*dt;
  P.x=clamp(P.x,WALL_L,WALL_R);
  if(P.y>=GY){
    P.y=GY;
    if(!wasG){
      if(pst==='plunge')plungeImpact();
      else if(pvy>420){dust(P.x,GY,7,0);sfx.land();shake(1.6);P.squash=Math.min(.6,pvy/1500);}
    }
    P.vy=0;P.onGround=true;P.jumps=0;P.coyote=.1;
  }else{P.onGround=false;P.coyote-=dt;}
  if(P.onGround&&Math.abs(P.vx)>30)P.runP+=Math.abs(P.vx)*dt*.055;
  // 刀尖轨迹
  if(st==='attack'&&P.t>P.atk.d.st&&P.t<P.atk.d.st+P.atk.d.ac+.02){
    const tip=swordTip();
    trail.push({x:tip.x,y:tip.y,a:1});
    if(trail.length>14)trail.shift();
  }
  updScarf(dt);
}
function swordPose(){
  // 返回 {swA, armA, lean, crouch}
  const st=P.state;let swA=REST,armA=.65,lean=0,crouch=0;
  if(st==='run'){swA=1.25+Math.sin(P.runP)*.12;armA=swA-.35;lean=.14;}
  else if(st==='air'){swA=P.vy<0?-.35:.5;armA=swA+.3;lean=.08;}
  else if(st==='dash'){swA=2.6;armA=2.1;lean=.42;}
  else if(st==='charge'){
    const lv=clamp(P.chgT/.75,0,1);
    swA=lerp(REST,-1.55,lv);armA=swA+.4;lean=-.15*lv;crouch=7*lv;
  }
  else if(st==='hurt'){swA=2.3;armA=1.9;lean=-.35;}
  else if(st==='plunge'){swA=1.57;armA=1.45;lean=.15;}
  else if(st==='plungeRec'){swA=1.3;armA=1.1;lean=.3;crouch=6;}
  else if(st==='attack'){
    const d=P.atk.d,t=P.t;
    if(t<d.st){const p=smooth(t/d.st);swA=lerp(REST,d.a0,p);lean=lerp(0,d.lean0,p);}
    else if(t<d.st+d.ac){const p=eoc((t-d.st)/d.ac);swA=lerp(d.a0,d.a1v,p);lean=lerp(d.lean0,d.lean1,p);}
    else{const p=Math.min(1,(t-d.st-d.ac)/d.rc*1.3);swA=lerp(d.a1v,REST,p);lean=lerp(d.lean1,0,p);}
    armA=swA*.55-.15;
  }
  return {swA,armA,lean,crouch};
}
function swordTip(){
  const p=swordPose();
  const shx=P.x+P.face*p.lean*10,shy=P.y-60+p.crouch;
  const hx=shx+P.face*Math.cos(p.armA)*13,hy=shy+Math.sin(p.armA)*13;
  return {x:hx+P.face*Math.cos(p.swA)*58,y:hy+Math.sin(p.swA)*58};
}

/* ============ 敌人 ============ */
const ETYPES={
  grunt:{hp:30,spd:180,r:20,h:64,dmg:12,wind:.45,strk:.14,rec:.6,rng:64,poise:4,col:'#251e2a',edge:'#3f2436',eye:'#ff3b30',w:5,alen:15},
  brute:{hp:130,spd:92,r:32,h:96,dmg:24,wind:.8,strk:.22,rec:.95,rng:104,poise:6,col:'#1e1826',edge:'#54222c',eye:'#ff7a1a',w:9,alen:23}
};
function Enemy(type,x){
  const T=ETYPES[type];
  this.type=type;this.x=x;this.y=GY;this.vx=0;this.vy=0;this.face=1;
  this.hp=T.hp;this.maxhp=T.hp;this.r=T.r;this.h=T.h;
  this.limbs={head:1,armF:1,armB:1,legF:1,legB:1};
  this.armN=2;this.legs=2;
  this.state='spawn';this.t=0;this.cd=rand(.6,1.2);this.flash=0;this.super=0;
  this.poiseN=0;this.walkP=rand(0,TAU);this.hDir=1;this.rot=0;this.dieSpin=0;
  this.lastHit=0;this.windD=T.wind;this.tk=false;this.sHit=false;this.dead=false;
}
Enemy.prototype.takeHit=function(o){
  if(this.hp<=0)return;
  this.hp-=o.dmg;this.flash=.13;this.hDir=o.dir;
  this.leanK=(this.leanK||0)+o.dir*10;
  let inter=this.super<=0&&!((this.type==='brute')&&(this.state==='windup'||this.state==='strike'));
  this.poiseN++;
  if(o.up){
    this.vy=o.up;this.vx=o.dir*(o.kb*.6+60);
    this.state='hurtAir';this.rot=0;this.dieSpin=o.dir*6;
  }else this.vx=o.dir*o.kb;
  if(inter&&!o.up){
    if(this.poiseN>=ETYPES[this.type].poise){
      this.poiseN=0;this.super=.9;this.state='stagger';
      this.t=o.heavy?.5:.38;
    }else if(this.state!=='stagger'){this.state='stagger';this.t=.3;}
  }
  if(o.sev&&this.hp>0&&Math.random()<o.sev+(this.hp<this.maxhp*.35?.18:0))this.breakLimb(null);
  if(this.hp<=0)this.die(o.dir);
};
Enemy.prototype.breakLimb=function(force){
  if(this.hp<=0&&force!=='head')return;
  const av=['armF','armB','legF','legB'].filter(k=>this.limbs[k]);
  let k=force;
  if(!k)k=av.length?av[rnd(0,av.length-1)]:null;
  if(!k)return;
  if(k==='head'&&!this.limbs.head)return;
  this.limbs[k]=0;limbBreaks++;
  limbsP.push(new LimbPiece(k,this));
  if(k==='axe')return;
  const bx=this.x+this.hDir*this.r*.3,by=this.y-this.h*.72;
  embers(bx,by,10);ichor(bx,by,this.hDir,7);smoke(bx,by,3);
  sfx.sever();freeze=Math.max(freeze,.11);shake(4);
  this.armN=this.limbs.armF+this.limbs.armB;
  this.legs=this.limbs.legF+this.limbs.legB;
  this.vx+=this.hDir*120;
  if(k==='head'){this.hp=0;this.die(this.hDir);}
};
Enemy.prototype.die=function(dir){
  if(this.state==='die')return;
  this.hp=0;this.state='die';this.t=0;
  this.vy=-300;this.vx=dir*300;this.hDir=dir;this.dieSpin=-dir*7;
  kills++;sfx.kill();slowT=.18;freeze=Math.max(freeze,.09);shake(6);
  embers(this.x,this.y-this.h*.5,16);smoke(this.x,this.y-this.h*.5,6);
  xflash(this.x,this.y-this.h*.5);
  if(this.limbs.head&&Math.random()<.5)this.breakLimb('head');
};
Enemy.prototype.update=function(dt){
  this.flash-=dt;this.super-=dt;this.cd-=dt;
  this.leanK=(this.leanK||0)*Math.exp(-8*dt);
  const T=ETYPES[this.type];
  const dx=P.x-this.x,adx=Math.abs(dx),dir=Math.sign(dx)||1;
  this.vy+=1300*dt;
  this.x+=this.vx*dt;this.y+=this.vy*dt;
  this.x=clamp(this.x,WALL_L-20,WALL_R+20);
  let g=false;
  if(this.y>=GY){
    this.y=GY;g=true;
    if(this.vy>260)dust(this.x,GY,4,0);
    this.vy=0;
    if(this.state==='hurtAir'){this.state='down';this.t=.62;this.vx*=.25;}
    if(this.state==='die'){
      corpses.push({type:this.type,x:this.x,face:this.hDir,limbs:this.limbs,life:4,h:this.h,big:this.type==='brute'});
      this.dead=true;return;
    }
  }
  const fr=g?Math.exp(-6*dt):Math.exp(-1.1*dt);
  if(this.state!=='strike'&&this.state!=='tackle')this.vx*=fr;
  switch(this.state){
    case 'spawn':this.t+=dt;if(this.t>=.55){this.state='chase';this.cd=rand(.5,1);}break;
    case 'chase':{
      this.face=dir;
      const spd=T.spd*(this.legs===2?1:this.legs===1?.45:.16);
      this.vx+=(dir*spd-this.vx)*Math.min(1,10*dt);
      this.walkP+=Math.abs(this.vx)*dt*.1;
      if(g&&this.type==='grunt'&&Math.random()<dt*.5&&adx>150)this.vy=-360;
      const useTk=this.armN===0||(this.type==='brute'&&!this.limbs.armF);
      const want=useTk?(adx<250):(adx<T.rng+20);
      if(this.cd<=0&&want&&P.hp>0){
        this.state='windup';this.t=0;this.tk=useTk;this.sHit=false;
        this.windD=(this.type==='brute'&&this.hp<T.hp*.4)?T.wind*.6:T.wind;
        this.windD=useTk?Math.min(this.windD,.32):this.windD;
        sfx.growl(this.type==='brute');this.vx*=.2;
      }
    }break;
    case 'windup':
      this.t+=dt;this.vx*=Math.exp(-8*dt);this.face=dir;
      if(this.t>=this.windD){
        this.t=0;this.sHit=false;
        if(this.tk){this.state='tackle';this.vx=this.face*430;sfx.swing(.45);}
        else{
          this.state='strike';
          this.vx=this.face*300;sfx.swing(.5);
          arcs.push({fol:2,e:this,a0:-1.9,a1:1.0,r0:14,r1:T.rng+16,life:.14,max:.14,face:this.face,col:'255,80,60'});
        }
      }break;
    case 'strike':
      this.t+=dt;
      if(!this.sHit&&this.t<T.strk&&P.hp>0){
        const hx=this.x+this.face*(T.rng*.55),hy=this.y-this.h*.55;
        if(Math.abs(P.x-hx)<T.rng*.6+16&&Math.abs((P.y-42)-hy)<48+this.h*.3){
          this.sHit=true;hurtPlayer(T.dmg,this.face);
        }
      }
      if(this.t>=T.strk+.06){this.state='rec';this.t=0;}break;
    case 'tackle':
      this.t+=dt;this.vx=this.face*430;
      if(Math.random()<dt*30)addP({k:'dust',x:this.x-this.face*this.r,y:GY-2,vx:-this.face*80,vy:rand(-50,-10),g:200,life:.3,max:.3,s:3});
      if(!this.sHit&&P.hp>0&&Math.abs(P.x-this.x)<this.r+20&&Math.abs(P.y-this.y)<64){
        this.sHit=true;hurtPlayer(this.armN>0?T.dmg*.6:T.dmg,this.face);
      }
      if(this.t>.5){this.state='rec';this.t=0;this.cd=rand(1.4,2.2);}break;
    case 'rec':
      this.t+=dt;this.vx*=Math.exp(-5*dt);
      if(this.t>=T.rec){this.state='chase';this.cd=rand(1,1.8);}break;
    case 'stagger':this.t-=dt;if(this.t<=0){this.state='chase';this.cd=rand(.6,1.2);}break;
    case 'down':this.t-=dt;if(this.t<=0){this.state='chase';this.cd=rand(.4,.9);}break;
    case 'hurtAir':break;
    case 'die':this.rot+=this.dieSpin*dt;this.vx*=Math.exp(-2*dt);break;
  }
  // 残端冒烟
  const miss=(this.limbs.armF?0:1)+(this.limbs.armB?0:1)+(this.limbs.legF?0:1)+(this.limbs.legB?0:1);
  if(miss>0&&Math.random()<dt*3*miss)embers(this.x+rand(-10,10),this.y-this.h*.55,1);
};

/* ============ 波次 ============ */
function nextWave(){
  wave++;
  const g=Math.min(2+Math.floor((wave-1)*.75),4);
  const b=wave>=2?Math.min(1+Math.floor((wave-2)/2),2):0;
  queue=[];
  for(let i=0;i<g;i++)queue.push('grunt');
  for(let i=0;i<b;i++)queue.push('brute');
  for(let i=queue.length-1;i>0;i--){const j=rnd(0,i);[queue[i],queue[j]]=[queue[j],queue[i]];}
  spawnT=.7;banner=1.8;bannerTxt='第 '+wave+' 波';
}
function updateSpawner(dt){
  if(mode!=='play'||P.hp<=0)return;
  if(queue.length){
    spawnT-=dt;
    if(spawnT<=0){
      spawnT=.75;
      const t=queue.pop(),side=Math.random()<.5?-1:1;
      let x=P.x+side*rand(430,660);
      if(x<WALL_L||x>WALL_R)x=P.x-side*rand(430,660);
      x=clamp(x,WALL_L+30,WALL_R-30);
      pending.push({x,t:.8,type:t});
    }
  }else if(!pending.length&&enemies.every(e=>e.hp<=0)){
    waveCd-=dt;
    if(waveCd<=0){waveCd=1.5;nextWave();}
  }
  for(let i=pending.length-1;i>=0;i--){
    const s=pending[i];s.t-=dt;
    if(s.t<=0){pending.splice(i,1);enemies.push(new Enemy(s.type,s.x));}
  }
}
function startGame(){reset();}
function reset(){
  enemies=[];corpses=[];limbsP=[];parts=[];arcs=[];trail=[];pending=[];queue=[];
  kills=0;totalHits=0;limbBreaks=0;combo=0;comboT=0;wave=0;waveCd=0;
  freeze=0;slowT=0;shakeAmp=0;hurtVig=0;banner=0;
  resetPlayer();mode='play';stopChargeSnd();nextWave();
}

/* ============ 主更新 ============ */
function update(dt){
  time+=dt;
  if(mode==='title'){
    P.vx*=Math.exp(-4*dt);updScarf(dt);
    for(const p of parts)p.life-=dt;
    parts=parts.filter(p=>p.life>0);
    camX+=(P.x-camX)*Math.min(1,3*dt);
    return;
  }
  if(freeze>0){freeze-=dt;shakeAmp*=Math.exp(-4*dt);return;}
  if(slowT>0)slowT-=dt;
  const sdt=dt*(slowT>0?.35:1);
  shakeAmp*=Math.exp(-7*dt);
  hurtVig=Math.max(0,hurtVig-dt*1.6);
  if(banner>0)banner-=dt;
  comboPop*=Math.exp(-8*dt);
  updatePlayer(sdt);
  for(const e of enemies)if(!e.dead)e.update(sdt);
  enemies=enemies.filter(e=>!e.dead);
  // 分离
  for(let i=0;i<enemies.length;i++)for(let j=i+1;j<enemies.length;j++){
    const a=enemies[i],b=enemies[j];
    const dx=b.x-a.x;
    if(Math.abs(dx)<a.r+b.r&&Math.abs(a.y-b.y)<50){
      const push=(a.r+b.r-Math.abs(dx))*.5*Math.sign(dx||1);
      b.x+=push;a.x-=push;
    }
  }
  for(const l of limbsP)l.update(sdt);
  limbsP=limbsP.filter(l=>l.life>0);
  if(limbsP.length>22)limbsP.splice(0,limbsP.length-22);
  for(const c of corpses)c.life-=sdt;
  corpses=corpses.filter(c=>c.life>0);
  for(const p of parts){
    p.life-=sdt;
    if(p.k==='spark'||p.k==='ichor'||p.k==='ember'||p.k==='dust'){p.vy+=(p.g||0)*sdt;p.x+=p.vx*sdt;p.y+=p.vy*sdt;}
    else if(p.k==='smoke'){p.x+=p.vx*sdt;p.y+=p.vy*sdt;p.s+=sdt*8;}
    else if(p.k==='suck'){p.x+=p.vx*sdt;p.y+=p.vy*sdt;}
  }
  parts=parts.filter(p=>p.life>0);
  for(const a of arcs)a.life-=sdt;
  arcs=arcs.filter(a=>a.life>0);
  for(const t of trail)t.a-=sdt*6.5;
  trail=trail.filter(t=>t.a>0);
  if(Math.random()<sdt*3)
    addP({k:'ember',x:camX+rand(-W/2,W/2),y:rand(80,560),vx:rand(-14,14),vy:rand(-30,-8),g:-14,life:rand(1.5,3),max:3,s:rand(1,2.2)});
  updateSpawner(sdt);
  camX+=(P.x+P.face*85-camX)*Math.min(1,6*sdt);
  camX=clamp(camX,W/2,WORLD-W/2);
}

function frame(t){
  acc+=Math.min(t-last,100)/1000;last=t;
  while(acc>=STEP){update(STEP);acc-=STEP;}

}
